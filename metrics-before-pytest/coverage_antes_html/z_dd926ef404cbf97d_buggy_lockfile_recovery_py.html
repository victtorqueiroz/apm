1"""Version-gated heal: recover from the v<=0.12.2 branch-ref cache 

2drift bug for users upgrading to a fixed APM release. 

3 

4Lockfiles generated by APM <= 0.12.2 may contain a phantom 

5``resolved_commit`` (the remote HEAD at the time of the buggy install) 

6while the on-disk content reflects an older commit. The content-hash 

7fallback in ``_resolve_download_strategy`` happily declares 

8``lockfile_match=True`` because both lockfile content_hash and disk 

9content reflect the OLD bytes, so the lie persists indefinitely until 

10upstream advances again. 

11 

12When we detect this combination (buggy lockfile generator + branch ref 

13+ content-hash-only match), force a one-time re-download. The 

14subsequent install records the current apm_version, so the trigger 

15does not re-fire on releases past 0.12.2. 

16 

17Shares ``exclusive_group="branch_drift"`` with 

18:class:`BranchRefDriftHeal` so the two never fire on the same dep -- 

19the always-on drift heal handles the case where the remote has 

20advanced; this heal handles the case where the remote has NOT advanced 

21but the lockfile is still corrupted from the buggy era. 

22""" 

23 

24from __future__ import annotations 

25 

26from typing import Any 

27 

28from .base import HealContext, HealMessageLevel 

29 

30# Lockfiles produced by these APM versions may contain phantom 

31# ``resolved_commit`` values for branch-ref deps (3-way drift bug). 

32# When such a lockfile is detected, this heal forces a one-time 

33# re-download of branch-ref deps that match via content-hash fallback 

34# only. See PR #1158. 

35_BUGGY_BRANCH_REF_DRIFT_VERSIONS = frozenset( 

36 { 

37 "0.10.0", 

38 "0.10.1", 

39 "0.10.2", 

40 "0.11.0", 

41 "0.11.1", 

42 "0.11.2", 

43 "0.12.0", 

44 "0.12.1", 

45 "0.12.2", 

46 } 

47) 

48 

49 

50def _is_buggy_lockfile_apm_version(existing_lockfile: Any) -> bool: 

51 """Return True when *existing_lockfile* was generated by an APM 

52 version known to suffer from the branch-ref cache-drift bug. 

53 

54 Conservative default (False when version is missing): the self-heal 

55 path is opt-in -- only trigger for lockfiles we can positively 

56 identify as affected. 

57 """ 

58 if existing_lockfile is None: 

59 return False 

60 recorded = getattr(existing_lockfile, "apm_version", None) 

61 if not recorded: 

62 return False 

63 return recorded in _BUGGY_BRANCH_REF_DRIFT_VERSIONS 

64 

65 

66class BuggyLockfileRecoveryHeal: 

67 name = "buggy_lockfile_recovery" 

68 order = 20 

69 exclusive_group = "branch_drift" 

70 

71 def applies(self, hctx: HealContext) -> bool: 

72 from apm_cli.models.apm_package import GitReferenceType 

73 

74 return ( 

75 hctx.lockfile_match 

76 and hctx.lockfile_match_via_content_hash_only 

77 and not hctx.update_refs 

78 and hctx.resolved_ref is not None 

79 and hctx.resolved_ref.ref_type == GitReferenceType.BRANCH 

80 and _is_buggy_lockfile_apm_version(hctx.existing_lockfile) 

81 ) 

82 

83 def execute(self, hctx: HealContext) -> None: 

84 hctx.lockfile_match = False 

85 hctx.ref_changed = True 

86 # In the corrupted state, lockfile.content_hash records OLD 

87 # bytes while disk content is also OLD. The forced re-download 

88 # will produce NEW bytes whose hash differs from the OLD 

89 # lockfile hash. Tell the FreshDependencySource that this 

90 # change is LEGITIMATE -- otherwise the supply-chain hard-block 

91 # would abort before we can repair the lockfile. 

92 hctx.add_bypass_key(hctx.package_key) 

93 hctx.emit( 

94 HealMessageLevel.WARN, 

95 f"Recovering {hctx.package_key} from " 

96 "branch-ref cache drift in lockfile generated by APM <= 0.12.2 " 

97 "-- forcing re-download to restore consistency. " 

98 "Upgrade APM (>= 0.13.0) to prevent recurrence.", 

99 )
