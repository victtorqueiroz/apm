17def get_effective_default_registry(data: dict, *, create_config: bool = True) -> str | None: 

18 """Return the effective default registry name for CLI shorthand routing. 

19 

20 Checks the project-level ``registries.default`` key from *data* (the raw 

21 ``apm.yml`` dict) first, then falls back to the user-level default 

22 configured via ``apm config set``. Returns ``None`` when no default is 

23 configured at either level.

25 Args:

27 reading the user-level default. 

28 """

31 

32 if not is_package_registry_enabled(create_config=create_config): 

33 return None 

34 except Exception: 

35 return None 

36 raw_regs = data.get("registries") 

37 if isinstance(raw_regs, dict): 37 ↛ 45line 37 didn't jump to line 45 because the condition on line 37 was always true

38 project_default = raw_regs.get("default") 

39 # Only honor a project default that names a configured registry -- 

40 # mirrors _parse_registries_block, which rejects an unconfigured 

41 # default at load time. Otherwise the CLI would bypass the GitHub 

42 # probe and write a dep the next manifest load cannot resolve. 

43 if project_default and project_default in {k for k in raw_regs if k != "default"}: 

44 return str(project_default)

51 # project registries block, and the user config has no knowledge of

122 """Build ``RegistryResolution`` for a registry dep on the cached install path. 

123 

124 Two sources, so the lockfile keeps ``resolved_url``, ``resolved_hash``, and 

125 ``version`` when the package is reused from disk instead of re-downloaded: 

126 

127 1. The resolver's ``last_resolutions`` map — filled when the BFS callback in 

128 ``resolve.py`` just downloaded this dep during dependency-graph resolution 

129 (e.g. first install with no prior lockfile). This path was the bug fix. 

130 2. The existing lockfile row — used on re-install when the tree is already on 

131 disk and was verified earlier (original phase-7 wiring). 

132 

133 If neither applies, a registry dep on the cached path would degrade to a 

134 v1-shaped lockfile entry (missing registry resolution fields). 

135 """ 

136 if dep_ref.source != "registry": 

137 return None 

138 hit = resolver_last_registry_resolution(ctx, dep_key) 

139 if hit is not None: 

140 return hit 

141 if dep_locked_chk and dep_locked_chk.resolved_url: 

142 from apm_cli.deps.registry.resolver import RegistryResolution 

143 

144 return RegistryResolution( 

145 resolved_url=dep_locked_chk.resolved_url, 

146 resolved_hash=dep_locked_chk.resolved_hash or "", 

147 version=dep_locked_chk.version or (dep_ref.reference or ""), 

148 ) 

149 return None
