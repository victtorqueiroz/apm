69def _all_target_prefixes() -> list[str]: 

70 """Union of pack prefixes for every real (deployable) target. 

71 

72 A target is considered deployable when ``detect_by_dir`` or 

73 ``auto_create`` is True; ``copilot-cowork`` (both False) is excluded 

74 because it is an opt-in pseudo-target. 

75 

76 Order is stable: KNOWN_TARGETS insertion order, with deduplication 

77 preserving first occurrence. This keeps downstream YAML deterministic. 

78 """ 

79 prefixes: list[str] = [] 

80 seen: set[str] = set() 

81 for profile in KNOWN_TARGETS.values(): 

82 if not (profile.detect_by_dir or profile.auto_create): 

83 continue 

84 for prefix in profile.effective_pack_prefixes: 

85 if prefix not in seen: 

86 seen.add(prefix) 

87 prefixes.append(prefix) 

88 return prefixes

91def _get_target_prefixes(target: str) -> list[str]: 

92 """Resolve pack-prefixes for a single target name. 

93 

94 Reads from ``KNOWN_TARGETS[target].effective_pack_prefixes``. Special 

95 cases: 

96 

97 * ``"all"`` -- union of every deployable target's prefixes (see 

98 :func:`_all_target_prefixes`). 

99 * ``"vscode"`` -- treated as an alias for ``"copilot"`` (both deploy 

100 to ``.github/``); kept for backward compatibility because 

101 ``vscode`` is a valid MCP-only adapter target_name. 

102 * Unknown targets -- fall back to the union, matching the previous 

103 behavior of falling through to the all-targets default. 

104 """ 

105 if target == "all": 

106 return _all_target_prefixes() 

107 if target == "vscode": 

108 return list(KNOWN_TARGETS["copilot"].effective_pack_prefixes) 

109 profile = KNOWN_TARGETS.get(target) 

110 if profile is None: 

111 return _all_target_prefixes() 

112 return list(profile.effective_pack_prefixes)

115def _filter_files_by_target( 

116 deployed_files: list[str], target: str | list[str] 

117) -> tuple[list[str], dict[str, str]]: 

118 """Filter deployed file paths by target prefix, with cross-target mapping. 

119 

120 When files are deployed under one target prefix (e.g. ``.github/skills/``) 

121 but the pack target is different (e.g. ``claude``), skills and agents are 

122 remapped to the equivalent target path. Commands, instructions, and hooks 

123 are NOT remapped -- they are target-specific. 

124 

125 *target* may be a single string or a list of strings. For a list, the 

126 union of all relevant prefixes and cross-target maps is used. 

127 

128 Returns: 

129 A tuple of ``(filtered_files, path_mappings)`` where *path_mappings* 

130 maps ``bundle_path -> disk_path`` for any file that was cross-target 

131 remapped. Direct matches have no entry in the dict. 

132 """ 

133 if isinstance(target, list): 

134 # Union all prefixes for the targets in the list 

135 prefixes: list[str] = [] 

136 seen_prefixes: set = set() 

137 for t in target: 

138 for p in _get_target_prefixes(t): 

139 if p not in seen_prefixes: 

140 seen_prefixes.add(p) 

141 prefixes.append(p) 

142 # Preserve every selected target's cross-map so one source path can 

143 # intentionally fan out to multiple target-specific destinations. 

144 cross_maps: list[tuple[str, str]] = [] 

145 for t in target: 

146 cross_maps.extend(_CROSS_TARGET_MAPS.get(t, {}).items()) 

147 else: 

148 prefixes = _get_target_prefixes(target) 

149 cross_maps = list(_CROSS_TARGET_MAPS.get(target, {}).items()) 

150 

151 direct = [f for f in deployed_files if any(f.startswith(p) for p in prefixes)] 

152 

153 path_mappings: dict[str, str] = {} 

154 if cross_maps: 

155 direct_set = set(direct) 

156 for f in deployed_files: 

157 if f in direct_set: 

158 continue 

159 for src_prefix, dst_prefix in cross_maps: 

160 if f.startswith(src_prefix): 

161 mapped = dst_prefix + f[len(src_prefix) :] 

162 # Containment guard: normalise the remapped path and 

163 # reject any result that escapes the destination prefix. 

164 # The ".." check catches root-level escapes; the prefix 

165 # check below is the authoritative containment barrier. 

166 normalised = posixpath.normpath(mapped) 

167 if ".." in normalised.split("/"): 167 ↛ 168line 167 didn't jump to line 168 because the condition on line 167 was never true

168 continue 

169 if not normalised.startswith(dst_prefix.rstrip("/")): 

170 continue 

171 # Preserve trailing slash (directory marker in lockfiles) 

172 if mapped.endswith("/") and not normalised.endswith("/"): 

173 normalised += "/" 

174 mapped = normalised 

175 if mapped not in direct_set: 

176 direct.append(mapped) 

177 direct_set.add(mapped) 

178 path_mappings[mapped] = f 

179 

180 return direct, path_mappings
