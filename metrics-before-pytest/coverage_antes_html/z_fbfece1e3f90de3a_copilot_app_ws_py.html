1"""Localhost WebSocket IPC client for the GitHub Copilot desktop App. 

2 

3The App binds a per-launch authenticated WebSocket on ``127.0.0.1:<port>`` 

4and writes the port and token to ``~/.copilot/run/ws.{port,token}`` 

5(mode 0o600 -- we re-check the token file's mode at read time and 

6refuse to use a group/other-readable token). Any local process that 

7can read those files can speak the full ``WsClientMessage`` dialect. 

8This module is the typed, synchronous, scope-limited Python client APM 

9uses to register a project from a filesystem path 

10(``create_project_from_path``). 

11 

12Scope: project registration only. 

13 

14Workflow rows are written via direct SQLite in 

15``copilot_app_db.deploy_workflow`` regardless of which path created the 

16project. This keeps lockfile ids stable (namespaced 

17``owner/pkg/stem``) instead of opaque server-side UUIDs, and removes 

18the need for paired ``create_workflow`` / ``update_workflow`` IPC. The 

19WS surface still earns its keep on the project side: the App runs full 

20discovery (owner/repo detection, default branch, account binding) and 

21the resulting project row is the one the webview already knows about, 

22which is how we avoid the white-screen failure mode where an 

23externally-written ``project_id`` doesn't resolve. When the App is 

24closed the WS surface is unavailable and the caller falls back to 

25direct SQLite project registration via ``copilot_app_project``. 

26 

27Synchronous by design 

28--------------------- 

29The install loop is fully synchronous; spinning up an asyncio loop 

30would force every primitive call site (``prompt_integrator``) into the 

31``asyncio.run`` ceremony with no benefit. We use 

32``websockets.sync.client.connect`` -- the official sync API shipped in 

33``websockets>=12`` -- which gives us blocking ``send``/``recv`` with 

34timeouts. 

35 

36Security posture 

37---------------- 

38We mirror the App's own webview behaviour: 

39 

40* Always present ``Origin: tauri://localhost``. The server rejects 

41 any other origin (``websocket.rs:489-525``). 

42* Token is supplied via the ``?token=<base64>`` query string the 

43 server expects. 

44* TCP probe in ``ws_available`` is bounded to 100ms so install never 

45 hangs when the port file is stale (App crashed without cleanup). 

46* Parser is permissive: tolerate unknown / added top-level fields so 

47 upstream additions don't break us. We extract only what we use. 

48"""

157def _scrub_token(text: str) -> str:

163 token is short-lived but still credential material, so we keep it

169def _token_file_mode_ok(path: Path) -> bool: 

170 """Return True iff the token file has no group/other permissions. 

171

173 tool, sync agent) widens the mode, anything readable by group or

175 server for as long as the App stays open. We treat any 

176 group/other bit as a refusal-to-read rather than warning-and-

183 files), so the POSIX-style ``& S_IRWXG | S_IRWXO`` test would 

184 spuriously reject every token file. We short-circuit to ``True`` 

185 on Windows; ACL hardening on Windows is out of scope for this 

186 module.

198def _read_creds() -> tuple[int, str] | None: 

199 """Return ``(port, token)`` from the App's run-files or ``None``. 

200 

201 Both files must exist and be readable; either missing means the App 

202 is not running (or never has been since boot). The token file's 

203 mode is verified to match the App's documented 0o600 posture -- 

204 if widened to group/other-readable we refuse rather than risk 

205 sending credential material extracted from a non-private file. 

206 We do not validate the token format itself; the server rejects 

207 invalid tokens at handshake. 

208 """ 

209 run = _run_dir() 

210 port_path = run / _PORT_FILE 

211 token_path = run / _TOKEN_FILE 

212 if not port_path.is_file() or not token_path.is_file(): 

213 return None 

214 if not _token_file_mode_ok(token_path): 

215 return None 

216 try: 

217 port = int(port_path.read_text(encoding="ascii").strip()) 

218 token = token_path.read_text(encoding="ascii").strip() 

219 except (OSError, ValueError): 

220 return None 

221 if not (0 < port < 65536) or not token: 

222 return None 

223 return port, token

226def ws_available() -> bool: 

227 """Quick liveness check: port file present AND TCP probe succeeds. 

228 

229 Bounded to ``_TCP_PROBE_TIMEOUT_S`` so a stale port file (App 

230 crashed without cleanup) never costs the user more than 100ms on 

231 install. Returns ``False`` on any failure -- callers fall through 

232 to the SQLite path silently. 

233 """ 

234 creds = _read_creds() 

235 if creds is None: 

236 return False 

237 port, _token = creds 

238 try: 

239 with socket.create_connection(("127.0.0.1", port), timeout=_TCP_PROBE_TIMEOUT_S): 

240 return True 

241 except (OSError, ValueError): 

242 return False

270class WsClient: 

271 """Synchronous WS-IPC client. 

272 

273 Use as a context manager so the underlying socket is closed even 

274 when an exception propagates:: 

275 

276 with WsClient() as client: 

277 project = client.create_project_from_path(repo_root) 

278 

279 The client exposes only ``create_project_from_path``: workflow 

280 rows are written via direct SQLite (``copilot_app_db``) regardless 

281 of which path created the project. This keeps lockfile ids 

282 namespaced and stable (see module docstring). 

283 

284 All public methods raise a ``WsError`` subtype on any failure; the 

285 caller in ``prompt_integrator._integrate_prompts_for_copilot_app`` 

286 catches ``WsError`` and falls through to the SQLite project 

287 registration path. 

288 """

318 def _connect(self) -> None: 

319 """Open the authenticated WS connection. 

320 

321 Raises: 

322 WsAppNotRunning: port/token files missing or TCP refused. 

323 WsAuthError: server rejected the token at handshake. 

324 WsProtocolError: any other handshake failure. 

325 """ 

326 # Lazy import: websockets is a hard dep but the import cost is 

327 # non-trivial and only matters when the App is running. 

328 try: 

329 from websockets.sync.client import connect 

330 except ImportError as exc: # pragma: no cover - hard dep 

331 raise WsError(f"websockets library unavailable: {exc}") from exc 

332 

333 creds = _read_creds() 

334 if creds is None: 

335 raise WsAppNotRunning("No ~/.copilot/run/ws.{port,token} files present") 

336 port, token = creds 

337 

338 url = f"ws://127.0.0.1:{port}/?token={token}" 

339 try: 

340 self._conn = connect( 

341 url, 

342 additional_headers={ 

343 "Origin": _ORIGIN_HEADER, 

344 "User-Agent": _USER_AGENT, 

345 }, 

346 open_timeout=_WS_HANDSHAKE_TIMEOUT_S, 

347 max_size=2**24, 

348 ) 

349 except Exception as exc: 

350 # Scrub the handshake URL's ``?token=<base64>`` before it 

351 # lands in our exception text -- the ``websockets`` library 

352 # echoes the full URL back in several of its InvalidStatus 

353 # / InvalidHandshake messages. 

354 raw = str(exc) 

355 msg = _scrub_token(raw) 

356 scrubbed_lower = msg.lower() 

357 # ``websockets`` raises InvalidStatus / InvalidHandshake for 

358 # HTTP-level rejections. 401/403 -> auth; everything else -> 

359 # generic protocol error. We string-match defensively so a 

360 # library-version bump doesn't break the branch. 

361 if "401" in msg or "403" in msg or "unauthor" in scrubbed_lower: 

362 raise WsAuthError(f"WS auth rejected: {msg}") from None 

363 if "refused" in scrubbed_lower or "connectionrefused" in scrubbed_lower: 

364 raise WsAppNotRunning(f"WS connection refused: {msg}") from None 

365 raise WsProtocolError(f"WS handshake failed: {msg}") from None

446 def create_project_from_path(self, path: Path) -> ProjectCreated: 

447 """Register *path* as a project via the App's own create flow. 

448 

449 The App's handler runs full discovery: folder probe, GitHub 

450 owner/repo detection, default-branch resolution, account 

451 binding. We get back the resolved ``project_id``. 

452 

453 ``was_created`` is inferred from the server's reply type: 

454 ``project_created`` -> ``was_created=True`` (new ``projects`` row), 

455 ``project_updated`` -> ``was_created=False`` (HIT on existing 

456 ``main_repo_path``). The restart hint downstream only fires on 

457 ``was_created=True``; on ``project_updated`` the webview already 

458 knows the row and no restart is needed. The "was new" flag rides 

459 on the message ``type`` rather than a dedicated field -- see 

460 ``_extract_project_fields`` for the parser. 

461 

462 Upstream issue github/github-app#5483 tracks adding a live 

463 broadcast so externally-inserted rows surface in the webview 

464 without a restart; until that lands, the hint is the user-visible 

465 signal that a manual restart wires the new project into the UI. 

466 """ 

467 self._send( 

468 { 

469 "type": "create_project_from_path", 

470 "path": str(path), 

471 } 

472 ) 

473 reply = self._await_typed_reply( 

474 expected={"project_created", "project_updated"}, 

475 ) 

476 project_id, main_repo_path, was_created = _extract_project_fields( 

477 reply, fallback_path=str(path) 

478 ) 

479 if not project_id: 

480 raise WsProtocolError(f"project_created reply missing id: {reply!r}") 

481 return ProjectCreated( 

482 project_id=project_id, 

483 was_created=was_created, 

484 main_repo_path=main_repo_path or str(path), 

485 )
