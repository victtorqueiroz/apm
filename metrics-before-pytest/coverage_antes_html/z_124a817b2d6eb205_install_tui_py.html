1"""Shared Live-region TUI controller for the install pipeline. 

2 

3PR #1116 / workstream B. 

4 

5A single ``InstallTui`` instance is opened by ``apm install`` and is 

6re-used across the resolve, download, integrate, and MCP-registry 

7phases. Per-phase code calls ``start_phase()`` once when the phase 

8boundary is crossed, then ``task_started()`` / ``task_completed()`` / 

9``task_failed()`` for every dep / server / artifact in flight. 

10 

11The Live region is **deferred** by 250 ms after open so that an 

12install that finishes from a warm cache or completes in <250 ms never 

13flashes a spinner. The ``should_animate()`` predicate gates the whole 

14controller on TTY capabilities and the ``APM_PROGRESS`` env knob. 

15 

16Notes for callers 

17----------------- 

18 

19* Always wrap the lifecycle in ``with ctx.tui:``. The context 

20 manager owns ``Live.stop()`` in the ``__exit__`` path and is the 

21 only safe place to tear the Live region down. 

22* When ``should_animate()`` is False (CI, dumb terminal, 

23 ``APM_PROGRESS=never``, ``--quiet``), every method on this class is 

24 a cheap no-op. Callers do NOT need to gate their calls. 

25* This module deliberately uses a single ASCII spinner 

26 (``spinner_name="line"`` => ``| / - \\``) and never emits emoji or 

27 Unicode box-drawing, to stay safe under Windows cp1252. 

28""" 

29

43# Defer the Live region for 250 ms after entering the context manager. 

44# Installs that finish under this threshold never paint a spinner. 

45_DEFER_SHOW_S: float = 0.250

61def should_animate() -> bool: 

62 """Return True iff the install pipeline should paint a Live region. 

63 

64 Resolution order (first match wins): 

65 

66 1. ``APM_PROGRESS=never`` or ``=quiet`` -- never animate. 

67 2. ``APM_PROGRESS=always`` -- always animate (intended for local 

68 debugging; CI MUST NOT set this). 

69 3. ``APM_PROGRESS=auto`` (default) -- animate iff the console is 

70 an interactive TTY AND ``TERM`` is not ``""`` / ``"dumb"`` AND 

71 ``CI`` is not truthy. 

72 

73 The function intentionally does NOT consult ``--quiet`` itself; 

74 the CLI front-end is responsible for setting ``APM_PROGRESS=quiet`` 

75 (or never instantiating ``InstallTui``) in that case. 

76 """ 

77 mode = os.environ.get("APM_PROGRESS", "auto").strip().lower() 

78 if mode in ("never", "quiet", "off", "0", "false", "no"): 

79 return False 

80 if mode in ("always", "on", "1", "true", "yes"): 

81 return True 

82 # mode == "auto" (or unrecognized -- treat as auto) 

83 if os.environ.get("CI", "").strip().lower() in ("1", "true", "yes"): 

84 return False 

85 if os.environ.get("TERM", "").strip().lower() in ("", "dumb"): 

86 return False 

87 c = _get_console() 

88 if c is None: 88 ↛ 89line 88 didn't jump to line 89 because the condition on line 88 was never true

89 return False 

90 try: 

91 return bool(getattr(c, "is_terminal", False)) and bool(getattr(c, "is_interactive", False)) 

92 except Exception: 

93 return False

101class InstallTui: 

102 """One Live region for the entire install lifecycle. 

103 

104 Public API (all calls are no-ops when the controller is disabled): 

105 

106 * ``__enter__`` / ``__exit__`` -- context-manager protocol. 

107 * ``start_phase(name, total)`` -- swap the aggregate progress bar 

108 to a fresh task for the named phase. 

109 * ``task_started(key, label)`` -- add ``label`` to the in-flight 

110 label set. Idempotent on label. 

111 * ``task_completed(key, milestone)`` -- remove labels matching 

112 ``key`` from the in-flight set, advance the phase bar, and 

113 optionally emit ``milestone`` as a non-transient line above the 

114 Live region. 

115 * ``task_failed(key, milestone)`` -- alias for ``task_completed``; 

116 callers are expected to format ``milestone`` with ``[x]``. 

117 * ``is_animating()`` -- True iff the Live region is currently 

118 visible (i.e. the defer threshold elapsed and the controller is 

119 enabled). Used by the resolving heartbeat to suppress its 

120 static line. 

121 """ 

122 

123 def __init__(self) -> None: 

124 self.console = _get_console() 

125 self._enabled: bool = should_animate() 

126 

127 # Lazily build the Rich primitives so non-animating installs 

128 # do not import or instantiate Progress / Live at all. 

129 self._aggregate: Any | None = None 

130 self._task_id: Any | None = None 

131 self._labels: list[str] = [] 

132 # Per-key tracking so task_completed(key) can drop the right 

133 # label even when callers use a human-readable label that does 

134 # not embed the dep key. Insertion-ordered for stable display. 

135 self._key_to_label: dict[str, str] = {} 

136 self._lock = threading.Lock() 

137 self._live: Any | None = None 

138 self._timer: threading.Timer | None = None 

139 # Sentinel to close a TOCTOU race between __exit__ on the main 

140 # thread and the deferred-start callback on the Timer thread: 

141 # if the timer is past cancel() but has not yet assigned _live, 

142 # _defer_start checks _shutdown after constructing Live and 

143 # before .start() so the region is never left running unowned. 

144 self._shutdown: bool = False 

145 

146 # -- Context-manager lifecycle ---------------------------------------- 

147 # 

148 # NOTE: This controller supports MULTIPLE enter/exit cycles on the 

149 # same instance. ``__exit__`` only tears down the Live region and 

150 # the deferred-show timer; ``_aggregate``, ``_labels``, and 

151 # ``_key_to_label`` survive so a follow-on ``__enter__`` can resume 

152 # rendering. The install pipeline relies on this: it wraps resolve 

153 # and the post-resolve body in two separate ``with`` blocks so an 

154 # early-exit "nothing to do" path can cleanly tear the Live region 

155 # down without losing phase state. 

156 

157 def __enter__(self) -> InstallTui: 

158 if self._enabled: 

159 with self._lock: 

160 self._shutdown = False 

161 self._timer = threading.Timer(_DEFER_SHOW_S, self._defer_start) 

162 self._timer.daemon = True 

163 self._timer.start() 

164 return self 

165 

166 def __exit__(self, *exc: Any) -> bool: 

167 # Set shutdown sentinel BEFORE cancel() so the Timer thread can 

168 # observe it and bail out even if it raced past the cancel. 

169 with self._lock: 

170 self._shutdown = True 

171 # ALWAYS cancel the deferred-start timer first; if cancel() 

172 # returns True the timer has not fired and we never built a 

173 # Live, so there is nothing to stop. 

174 if self._timer is not None: 

175 with contextlib.suppress(Exception): 

176 self._timer.cancel() 

177 self._timer = None 

178 if self._live is not None: 

179 # Rich teardown is best-effort; never let Live cleanup 

180 # mask a real install error propagating from the body. 

181 with contextlib.suppress(Exception): 

182 self._live.stop() 

183 self._live = None 

184 return False # do not suppress exceptions

188 def _build_aggregate(self) -> Any: 

189 """Lazily construct the Rich ``Progress`` primitive. 

190 

191 Uses a custom ASCII bar column instead of Rich's default ``BarColumn`` 

192 because the latter renders Unicode block-drawing glyphs (U+2501 etc) 

193 that violate the cp1252 ASCII-only output contract. 

194 """ 

195 from rich.progress import ( 

196 Progress, 

197 ProgressColumn, 

198 SpinnerColumn, 

199 TaskProgressColumn, 

200 TextColumn, 

201 TimeElapsedColumn, 

202 ) 

203 from rich.text import Text 

204 

205 class _AsciiBarColumn(ProgressColumn): 

206 """ASCII-only progress bar: ``[####........]``.""" 

207 

208 def __init__(self, bar_width: int = 28) -> None: 

209 super().__init__() 

210 self._bar_width = bar_width 

211 

212 def render(self, task: Any) -> Any: 

213 pct = task.percentage if task.total else 0.0 

214 filled = round(self._bar_width * (pct / 100.0)) 

215 filled = max(0, min(self._bar_width, filled)) 

216 bar = "#" * filled + "." * (self._bar_width - filled) 

217 return Text(f"[{bar}]") 

218 

219 return Progress( 

220 _AsciiBarColumn(bar_width=28), 

221 TaskProgressColumn(), 

222 TextColumn("{task.fields[phase]}"), 

223 SpinnerColumn(spinner_name="line"), # ASCII: | / - \ 

224 TimeElapsedColumn(), 

225 console=self.console, 

226 refresh_per_second=_REFRESH_HZ, 

227 transient=True, 

228 ) 

229 

230 def _defer_start(self) -> None: 

231 """Timer callback: open the Live region after the defer window.""" 

232 try: 

233 with self._lock: 

234 if self._shutdown or self._live is not None: 

235 return 

236 from rich.console import Group 

237 from rich.live import Live 

238 

239 if self._aggregate is None: 239 ↛ 241line 239 didn't jump to line 241 because the condition on line 239 was always true

240 self._aggregate = self._build_aggregate() 

241 live = Live( 

242 Group(self._aggregate, self._labels_renderable()), 

243 console=self.console, 

244 refresh_per_second=_REFRESH_HZ, 

245 transient=True, 

246 redirect_stdout=False, 

247 redirect_stderr=False, 

248 ) 

249 # Re-check shutdown sentinel under the lock just before 

250 # publishing the Live reference and starting it. If __exit__ 

251 # set _shutdown after our first check (race window), bail 

252 # out before .start() so the region is never left orphaned. 

253 with self._lock: 

254 if self._shutdown: 254 ↛ 255line 254 didn't jump to line 255 because the condition on line 254 was never true

255 return 

256 self._live = live 

257 self._live.start(refresh=True) 

258 except Exception: 

259 # Defensive: a Live failure must NEVER take the install 

260 # down with it. Disable the controller and continue. 

261 self._enabled = False 

262 self._live = None 

263 

264 def _labels_renderable(self) -> Any: 

265 """Render the in-flight label list (called under the live refresh).""" 

266 from rich.text import Text 

267 

268 with self._lock: 

269 if not self._labels: 

270 return Text("") 

271 visible = self._labels[:_MAX_VISIBLE_LABELS] 

272 head = " > " + " ".join(visible) 

273 extra = len(self._labels) - len(visible) 

274 if extra > 0: 

275 head += f" ... and {extra} more" 

276 return Text(head, style="cyan") 

277 

278 def _refresh_group(self) -> None: 

279 """Re-render the Live group (aggregate bar + labels).""" 

280 if self._live is None: 

281 return 

282 try: 

283 from rich.console import Group 

284 

285 self._live.update( 

286 Group(self._aggregate, self._labels_renderable()), 

287 refresh=False, 

288 ) 

289 except Exception: 

290 pass

294 def is_animating(self) -> bool: 

295 """True iff the Live region is currently painted.""" 

296 return self._enabled and self._live is not None

298 def start_phase(self, name: str, total: int | None) -> None: 

299 """Swap the aggregate bar to a fresh task for ``name``. 

300 

301 ``total`` is the count of task units in this phase (deps to 

302 download, integrators to run, etc.). ``None`` is treated as 

303 ``1`` so the bar is well-formed but never completes from 

304 ``advance()`` calls alone. 

305 """ 

306 if not self._enabled: 

307 return 

308 if self._aggregate is None: 

309 self._aggregate = self._build_aggregate() 

310 if self._task_id is not None: 

311 with contextlib.suppress(Exception): 

312 self._aggregate.remove_task(self._task_id) 

313 self._task_id = None 

314 # Clear stale labels from prior phase so the active-set list does 

315 # not bleed across phase boundaries. 

316 with self._lock: 

317 self._key_to_label.clear() 

318 self._labels = [] 

319 self._task_id = self._aggregate.add_task( 

320 "", total=(total if total and total > 0 else 1), phase=name 

321 ) 

322 self._refresh_group() 

323 

324 def task_started(self, key: str, label: str) -> None: 

325 """Add ``label`` to the in-flight label list (de-duped on key).""" 

326 if not self._enabled: 

327 return 

328 with self._lock: 

329 if key not in self._key_to_label: 

330 self._key_to_label[key] = label 

331 if label not in self._labels: 

332 self._labels.append(label) 

333 self._refresh_group() 

334 

335 def task_completed(self, key: str, milestone: str | None = None) -> None: 

336 """Drop the label registered for ``key``, advance the phase bar. 

337 

338 If ``milestone`` is provided, it is printed above the Live 

339 region as a permanent line (the Live region is transient and 

340 will be torn down at exit). 

341 """ 

342 if not self._enabled: 

343 return 

344 with self._lock: 

345 label = self._key_to_label.pop(key, None) 

346 if label is not None: 

347 # A label may legitimately be shared by two keys; only 

348 # drop it from the visible list when no other key is 

349 # still using it. 

350 if label not in self._key_to_label.values(): 

351 self._labels = [lbl for lbl in self._labels if lbl != label] 

352 if self._aggregate is not None and self._task_id is not None: 

353 with contextlib.suppress(Exception): 

354 self._aggregate.advance(self._task_id, 1) 

355 if milestone and self._live is not None: 

356 # Rich's Console acquires its own internal lock here; do 

357 # NOT wrap with self._lock (would deadlock with the 

358 # refresh thread). 

359 with contextlib.suppress(Exception): 

360 self._live.console.print(milestone) 

361 self._refresh_group() 

362 

363 def task_failed(self, key: str, milestone: str | None = None) -> None: 

364 """Same lifecycle as :meth:`task_completed`; caller marks failure.""" 

365 self.task_completed(key, milestone)
