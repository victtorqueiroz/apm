142def _build_expected_install_paths(declared_deps, lockfile, apm_modules_dir: Path) -> set: 

143 """Build expected package paths under *apm_modules_dir*. 

144 

145 Combines direct deps (from ``apm.yml``) with transitive deps 

146 (depth > 1 from ``apm.lock``), using ``get_install_path()`` for 

147 consistency with how packages are actually installed. 

148 """ 

149 expected = set() 

150 for dep in declared_deps: 

151 install_path = dep.get_install_path(apm_modules_dir) 

152 try: 

153 relative_path = install_path.relative_to(apm_modules_dir) 

154 expected.add(relative_path.as_posix()) 

155 except ValueError: 

156 expected.add(str(install_path)) 

157 

158 if lockfile: 

159 for dep in lockfile.get_package_dependencies(): 

160 if dep.depth is not None and dep.depth > 1: 

161 dep_ref = dep.to_dependency_ref() 

162 install_path = dep_ref.get_install_path(apm_modules_dir) 

163 try: 

164 relative_path = install_path.relative_to(apm_modules_dir) 

165 expected.add(relative_path.as_posix()) 

166 except ValueError: 

167 pass 

168 return expected

171def _expand_with_ancestors( 

172 paths: Iterable[str], installed: Iterable[str] | None = None 

173) -> set[str]: 

174 """Expand a set of expected paths to include ancestor prefixes. 

175 

176 Given ``{"owner/repo/.apm/skills/my-skill"}``, returns a set containing 

177 the original path plus all intermediate path prefixes with 2+ segments 

178 (e.g., ``"owner/repo"``, ``"owner/repo/.apm"``, 

179 ``"owner/repo/.apm/skills"``, plus the original 

180 ``"owner/repo/.apm/skills/my-skill"``). 

181 This allows O(1) membership checks when determining whether a scanned 

182 directory is an ancestor of an expected package path. 

183 

184 Ancestor expansion exists because a subdirectory dependency 

185 (``git: owner/repo, path: .apm/skills/x``) is installed by cloning the 

186 entire repo to ``apm_modules/owner/repo/``. Intermediate filesystem 

187 directories created by that clone are required parts of the install -- 

188 not stale leftovers. 

189 

190 Real-orphan safety: when *installed* is supplied, an ancestor that 

191 matches one of the installed paths is NOT added to the expansion 

192 unless that path is also directly declared in *paths*. Callers should 

193 pass only the subset of installed paths that look like *real 

194 standalone packages* (i.e., directories that ship their own 

195 ``apm.yml`` or ``SKILL.md``) -- not filesystem intermediaries (which typically have 

196 only a ``.apm/`` subtree from a cloned subdir dep). This preserves 

197 orphan detection for the case where a user has a genuinely orphaned 

198 ``owner/repo`` package on disk alongside a declared sibling 

199 subdirectory dep (``owner/repo/.apm/skills/foo``): only filesystem 

200 intermediaries are suppressed, never real installed packages. 

201 

202 Security contract -- ancestor depth cap: ``get_install_path()`` 

203 anchors installs at the 2-segment repo root (GitHub) or 3-segment 

204 root (ADO). Anything deeper is a filesystem-intermediary path 

205 (``.apm/``, ``skills/``, ...) that ``_scan_installed_packages`` 

206 skips, so emitting ancestors past depth 3 would only widen the 

207 orphan-suppression surface without serving any real lookup. The 

208 loop is therefore capped at depth 3 (``min(4, len(parts))``), which 

209 bounds the number of paths an attacker-influenced ``apm.yml`` dep 

210 declaration can hide from orphan detection. If the install strategy 

211 ever grows deeper roots, lift this cap and document the new 

212 invariant here. 

213 

214 Traversal guard: any input path that fails 

215 :func:`apm_cli.utils.path_security.validate_path_segments` (which 

216 rejects both ``.`` and ``..`` segments after backslash 

217 normalisation) is kept in the result as-is (membership check) but 

218 produces no ancestors. Routing through the canonical guard -- 

219 rather than a hand-rolled ``".." in parts`` check -- ensures 

220 single-dot segments (``owner/./repo``) are also caught and keeps 

221 the project's path-validation contract centralised. 

222 """ 

223 materialized = list(paths) 

224 materialized_set = set(materialized) 

225 expanded = set(materialized) 

226 installed_set = set(installed) if installed is not None else set() 

227 for p in materialized: 

228 try: 

229 validate_path_segments(p, context="ancestor expansion") 

230 except PathTraversalError: 

231 continue 

232 # Normalise backslashes so Windows-style tokens split into the 

233 # same parts as POSIX inputs for the depth-capped loop below. 

234 normalised = p.replace("\\", "/") 

235 parts = normalised.split("/") 

236 # Cap at depth 3 -- the ADO install-root depth -- to bound the 

237 # ancestor-suppression surface (see security contract above). 

238 for i in range(2, min(4, len(parts))): 

239 ancestor = "/".join(parts[:i]) 

240 # Do not mask a real installed package via ancestor expansion; 

241 # only filesystem intermediaries should be added. A real 

242 # installed package that is also directly declared remains in 

243 # expanded via materialized_set. 

244 if ancestor in installed_set and ancestor not in materialized_set: 

245 continue 

246 expanded.add(ancestor) 

247 return expanded

250def _standalone_installed_packages( 

251 installed: Iterable[str], apm_modules_dir: Path, lockfile=None 

252) -> list: 

253 """Filter *installed* to entries that look like real standalone packages. 

254 

255 Determination order (tamper-evident first): 

256 

257 1. Path appears as a dependency key in *lockfile* -- the canonical 

258 record of what APM installed. The lockfile is integrity-checked 

259 and not forgeable by dropping/omitting files in ``apm_modules/``. 

260 2. Fallback: path has its own ``apm.yml`` or ``SKILL.md`` file. Used when the lockfile 

261 is absent (older installs / fresh checkouts) or does not list 

262 the key. A directory with only a ``.apm/`` marker is treated as 

263 a filesystem intermediary, not a standalone package. 

264 

265 Package markers preserve standalone orphan detection even when a 

266 declaration points at a subdirectory of a removed package root. 

267 

268 Failure mode: only narrowly-typed shape errors against 

269 ``lockfile.dependencies`` (``AttributeError`` / ``TypeError`` / 

270 ``KeyError``) are absorbed and degrade to the package-marker 

271 fallback. Any other exception (e.g. lockfile parse / I/O failure) 

272 propagates so the outer caller can decide whether to log or fail 

273 closed -- preventing a corrupted or attacker-crafted lockfile from 

274 silently disabling the tamper-evident standalone check. 

275 """ 

276 lockfile_keys: set[str] = set() 

277 if lockfile is not None: 

278 try: 

279 for dep_key in lockfile.dependencies: 

280 if dep_key: 280 ↛ 279line 280 didn't jump to line 279 because the condition on line 280 was always true

281 lockfile_keys.add(dep_key) 

282 except (AttributeError, TypeError, KeyError): 

283 lockfile_keys = set() 

284 standalone: list = [] 

285 for p in installed: 

286 if p in lockfile_keys: 

287 standalone.append(p) 

288 continue 

289 if (apm_modules_dir / p / APM_YML_FILENAME).exists() or ( 

290 apm_modules_dir / p / SKILL_MD_FILENAME 

291 ).is_file(): 

292 standalone.append(p) 

293 return standalone

296def _check_orphaned_packages(): 

297 """Check for packages in apm_modules/ that are not declared in apm.yml or apm.lock. 

298 

299 Considers both direct dependencies (from apm.yml) and transitive dependencies 

300 (from apm.lock) as expected packages, so transitive deps are not falsely 

301 flagged as orphaned. 

302 

303 Returns: 

304 List[str]: List of orphaned package names in org/repo or org/project/repo format 

305 """ 

306 try: 

307 if not Path(APM_YML_FILENAME).exists(): 

308 return []

311 if not apm_modules_dir.exists():

331def _find_orphaned_packages(installed: Iterable[str], expected: set[str]) -> list[str]: 

332 """Select roots unrelated to the retained dependency graph. 

333 

334 Keep both bundled descendants and whole roots containing needed children. 

335 Ancestor prefixes protect only those containing roots, not their siblings. 

336 Unlike dependency-list classification, deletion must retain recognized 

337 ancestors at every depth, including aliases and hidden subdirectories. 

338 """ 

339 roots = set() 

340 retained = set(expected) 

341 for path in expected: 

342 try: 

343 validate_path_segments(path, context="orphan selection") 

344 except PathTraversalError: 

345 # Invalid tokens protect only an exact match, never a wider subtree. 

346 continue 

347 normalized = path.replace("\\", "/") 

348 roots.add(normalized) 

349 parts = normalized.split("/") 

350 retained.update("/".join(parts[:depth]) for depth in range(1, len(parts) + 1)) 

351 

352 orphaned = [] 

353 for path in installed: 

354 normalized = path.replace("\\", "/") 

355 if normalized in retained: 

356 continue 

357 parts = normalized.split("/") 

358 if any("/".join(parts[:depth]) in roots for depth in range(1, len(parts))): 

359 continue 

360 orphaned.append(path) 

361 return sorted(orphaned)

382def resolve_requested_packages( 

383 packages: tuple[str, ...] | list[str], 

384 all_deps: list, 

385) -> list[str] | None: 

386 """Map requested package tokens to canonical dependency keys. 

387 

388 The install engine matches ``only_packages`` by ``DependencyReference`` 

389 identity (e.g. ``owner/repo``), so short names like ``compliance-rules`` 

390 must be normalised to their canonical key before the engine is called. 

391 

392 Args: 

393 packages: Raw tokens from the CLI ``[PACKAGES]...`` argument. 

394 all_deps: Declared dependencies (direct + dev) to match against. 

395 

396 Returns: 

397 An order-preserving, de-duplicated list of canonical keys, or 

398 ``None`` when *packages* is empty (meaning "refresh everything"). 

399 

400 Raises: 

401 UnknownPackageError: when a token matches no declared dependency. 

402 """ 

403 if not packages: 

404 return None 

405 

406 token_to_canonical: dict[str, str] = {} 

407 for dep in all_deps: 

408 canonical_key = dep.get_unique_key() or dep.repo_url or dep.get_display_name() 

409 tokens = {canonical_key, dep.get_display_name(), dep.repo_url} 

410 if getattr(dep, "alias", None): 

411 tokens.add(dep.alias) 

412 parts = dep.repo_url.split("/") 

413 if len(parts) >= 2: 413 ↛ 415line 413 didn't jump to line 415 because the condition on line 413 was always true

414 tokens.add(parts[-1]) 

415 for token in tokens: 

416 if token and token not in token_to_canonical: 

417 token_to_canonical[token] = canonical_key 

418 

419 resolved: list[str] = [] 

420 seen: set[str] = set() 

421 for pkg in packages: 

422 canonical = token_to_canonical.get(pkg) 

423 if not canonical: 

424 available = [dep.get_display_name() for dep in all_deps] 

425 raise UnknownPackageError(pkg, available) 

426 if canonical not in seen: 

427 seen.add(canonical) 

428 resolved.append(canonical) 

429 return resolved

688def _create_minimal_apm_yml(config, plugin=False, target_path=None): 

689 """Create minimal apm.yml file with auto-detected metadata. 

690 

691 Args: 

692 config: dict with name, version, description, author keys. 

693 Optional 'target' key: CSV string of targets to pin. 

694 plugin: if True, include a devDependencies section. 

695 target_path: explicit file path to write (defaults to cwd/apm.yml). 

696 """ 

697 # Create minimal apm.yml structure 

698 apm_yml_data = { 

699 "name": config["name"], 

700 "version": config["version"], 

701 "description": config["description"], 

702 "author": config["author"], 

703 } 

704 

705 # Add targets field if present in config (plural list form -- canonical). 

706 # Older callers may still pass a singular CSV "target" string; honor that 

707 # for backwards compatibility but normalise on disk to plural list form. 

708 if config.get("targets"): 

709 apm_yml_data["targets"] = list(config["targets"]) 

710 elif config.get("target"): 

711 raw = config["target"] 

712 if isinstance(raw, list): 

713 apm_yml_data["targets"] = list(raw) 

714 else: 

715 apm_yml_data["targets"] = [t.strip() for t in str(raw).split(",") if t.strip()] 

716 

717 apm_yml_data["dependencies"] = {"apm": [], "mcp": []} 

718 # Issue #887: scaffold with explicit consent for local content 

719 # deployment so day-2 audit doesn't surprise the maintainer with 

720 # an "includes not declared" advisory the moment they drop a 

721 # primitive in .apm/. Override with an explicit path list to 

722 # gate what gets deployed. 

723 apm_yml_data["includes"] = "auto" 

724 

725 if plugin: 

726 apm_yml_data["devDependencies"] = {"apm": []} 

727 

728 apm_yml_data["scripts"] = {} 

729 

730 # Write apm.yml 

731 from ..utils.yaml_io import dump_yaml 

732 

733 out_path = target_path or APM_YML_FILENAME 

734 dump_yaml(apm_yml_data, out_path) 

735 

736 # Post-process: add target comment header 

737 out_file = Path(out_path) 

738 content = out_file.read_text(encoding="utf-8") 

739 

740 if "targets" in apm_yml_data: 

741 from apm_cli.core.target_catalog import manifest_target_names 

742 

743 # Insert comment before the targets: line 

744 accepted_targets = ", ".join(sorted(manifest_target_names())) 

745 targets_comment = ( 

746 "# Which agent platforms to deploy to.\n" 

747 "# Resolution order: --target flag > this field > auto-detect from filesystem.\n" 

748 f"# Accepted values: {accepted_targets}\n" 

749 ) 

750 content = content.replace("targets:", targets_comment + "targets:", 1) 

751 else: 

752 # Insert commented-out skeleton before dependencies: 

753 skeleton = ( 

754 "# Which agent platforms to deploy to (uncomment to pin):\n" 

755 "# targets:\n" 

756 "# - copilot\n" 

757 "# - claude\n" 

758 ) 

759 content = content.replace("dependencies:", skeleton + "\ndependencies:", 1) 

760 

761 out_file.write_text(content, encoding="utf-8")

764def _find_apm_yml(start: Path | None = None) -> Path | None: 

765 """Walk parent directories from ``start`` (or cwd) to find ``apm.yml``. 

766 

767 Matches the npm / cargo / poetry ergonomic: a developer running 

768 an ``apm`` command from a subdirectory of their project (``src/``, 

769 ``docs/``, ``scripts/``) finds the manifest and operates on it. 

770 

771 The walk stops at the filesystem root or when an ``apm.yml`` is 

772 found, whichever comes first. Returns the absolute path to the 

773 ``apm.yml`` file when found; ``None`` when no project root is 

774 discoverable from ``start`` upward. 

775 """ 

776 cwd = (start or Path.cwd()).resolve() 

777 for candidate in (cwd, *cwd.parents): 

778 manifest = candidate / "apm.yml" 

779 if manifest.is_file(): 

780 return manifest 

781 return None
