68# Credential variable denylist -- these must never be expanded into HTTP 

69# headers or leaked to script subprocesses. The credential token must end 

70# the name, but we also accept a trailing plural ``S`` and an optional 

71# ``_ID``/``_IDS`` qualifier so real-world families are caught: 

72# - plurals: ...TOKENS, ...KEYS, ...SECRETS, ...CREDENTIALS, ...PATS 

73# - qualified: AWS_ACCESS_KEY_ID, ..._KEY_IDS 

74# - canonical: GOOGLE_APPLICATION_CREDENTIALS (CREDENTIAL + plural S) 

75# - key passphrases: GPG_PASSPHRASE, SSH_KEY_PASSPHRASE (PASSPHRASE is the 

76# same secret class as PASSWORD/PASSWD/PWD, which are already covered) 

77# - bare PASS shorthand: DB_PASS, MYSQL_PASS, REDIS_PASS, SMTP_PASS, 

78# ROOT_PASS (ubiquitous in docker-compose / 12-factor stacks). The bare 

79# ``PASS`` token is anchored to a ``_``-or-start boundary (``(?:^|_)PASS``) 

80# so the real ``*_PASS`` family is swept WITHOUT catching the common- 

81# English suffix words SURPASS / BYPASS / COMPASS / TRESPASS / PASSAGE. 

82# - one-time passcodes: MFA_PASSCODE, VPN_PASSCODE, *_PASSCODE (a dedicated 

83# ``PASSCODE`` token; the ``(?:^|_)PASS`` arm cannot reach it because the 

84# trailing ``CODE`` blocks the end anchor). Same password secret class. 

85# Benign ``*_CODE`` names (BARCODE, ZIPCODE, QRCODE, STATUS_CODE, 

86# COUNTRY_CODE, ERROR_CODE) never contain ``PASSCODE`` and stay benign. 

87# - bare JWT bearer tokens: JWT, ACCESS_JWT, REFRESH_JWT, *_JWT (a trailing- 

88# anchored ``JWT`` token). The ``eyJ...`` value matches no structural 

89# masker, so a bare *_JWT name is the only signal; benign LEADING-JWT 

90# config names (JWT_ALGORITHM, JWT_ISSUER, JWT_AUDIENCE) keep a non-token 

91# tail and so stay benign. JWT_SECRET is already swept via SECRET. 

92# - wallet seed phrases: MNEMONIC, *_MNEMONIC, *_SEED_PHRASE, *_RECOVERY_PHRASE, 

93# *_BACKUP_PHRASE (the hardhat / foundry / truffle deploy lifecycle secret -- 

94# a single token that, if leaked, drains a wallet; same secret class as 

95# PASSWORD). RECOVERY_PHRASE / BACKUP_PHRASE are the MetaMask / hardware-wallet 

96# spellings; the curated *_SEED wallet names (WALLET_SEED, MASTER_SEED, 

97# DERIVATION_SEED) live in _CREDENTIAL_BLOB_NAMES so a bare ``SEED`` token does 

98# NOT sweep the legitimate RNG seeds a build needs (PYTHONHASHSEED, RANDOM_SEED). 

99# The trailing ``S?`` does not over-match unrelated names (e.g. PATH keeps 

100# a stray ``H`` after PAT and so never matches; TRACE_ID has no credential 

101# token before ``_ID`` and is left alone). The trailing rotation tail 

102# ``(?:_?(?:OLD|NEW|PREV|CURRENT))?(?:_?V[0-9]+)?[_0-9]*`` lets an ENUMERATED / 

103# ROTATED credential keep matching even when a digit, an underscore, a rotation 

104# word (DB_PASS_OLD, PASSWORD_NEW), or a version tag (DB_PASSV2, API_KEYV2) now 

105# follows the token -- without it a single trailing word/digit defeats the whole 

106# sweep. The rotation words and ``V<n>`` tag only ever apply AFTER a credential 

107# token, so RENEW / PREVIEW / CURRENT alone (no token) stay benign, and 

108# SURPASS / PASSAGE (letters after PASS) stay benign via the ``(?:^|_)PASS`` 

109# boundary anchor. 

110# The looser tokens (KEY, PAT, ...) keep their suffix-match behaviour as a 

111# deliberate fail-safe: over-redacting a non-secret env var is harmless, 

112# under-redacting a secret is not. 

113# The trailing ENCODING tail closes a real, widely-deployed CI convention: a 

114# whole secret (a GCP service-account JSON, a TLS private key, a JWT signing 

115# secret, an RFC-6238 TOTP/MFA base32 seed, a Solana/web3 base58 key) is 

116# serialized or base-encoded into ONE single-line env var named ``<x>_BASE64`` / 

117# ``<x>_B64`` / ``<x>_BASE32`` / ``<x>_JSON`` / ``<x>_BASE58`` so it survives a 

118# flat secret store -- e.g. ``GCP_SA_KEY_BASE64``, ``TLS_PRIVATE_KEY_BASE64``, 

119# ``JWT_SECRET_B64``, ``GOOGLE_CREDENTIALS_JSON``, ``GCP_CREDENTIALS_YAML``, 

120# ``SOLANA_PRIVATE_KEY_BASE58``, ``TOTP_SECRET_BASE32``. The tail therefore spans 

121# three families: binary base encodings (BASE64/BASE32/BASE58/BASE62/B64/B32/HEX/ 

122# ASCII85/A85/Z85/URLSAFE), key armor (PEM/DER/ASC), and raw structured-text 

123# serialization (JSON/YAML/YML/TOML -- how GCP/terraform inline a credentials 

124# blob). The credential token (KEY/TOKEN/SECRET/CREDENTIAL/AUTHORIZATION) is an 

125# INFIX and the name ENDS in the benign tail, which the bare suffix anchor could 

126# not express. The tail only ever applies AFTER a credential token, so a 

127# TOKEN-LESS asset (IMAGE_BASE64, LOGO_B64, CONFIG_JSON, PACKAGE_JSON, COLOR_HEX) 

128# never matches and still reaches the child env. 

129# ``(?:^|_)COOKIE`` is start-anchored like ``PASS``: a session/auth COOKIE is a 

130# bearer credential (SESSION_COOKIE / AUTH_COOKIE / COOKIE / COOKIES) but the 

131# benign cookie *config* a script reads (COOKIE_DOMAIN / COOKIE_NAME / 

132# COOKIE_PATH / COOKIE_SECURE -- COOKIE is a PREFIX there) must survive, so the 

133# token only matches when COOKIE is the trailing segment.

260 """True if *name* conventionally holds a credential value. 

261

303def _redact_secrets(text: str) -> str: 

304 """Mask any denylisted env-var *values* appearing in script output. 

305 

306 Scripts frequently echo their environment; without this, a command 

307 that prints ``$ANALYTICS_TOKEN`` would persist the cleartext secret 

308 into ``~/.apm/logs/scripts.log``. We replace raw occurrences of every 

309 denylisted variable's value with ``[REDACTED]``. 

310 

311 Replacement is a raw substring match (not boundary-aware): the value 

312 is a KNOWN secret, so it must be masked even when glued to adjacent 

313 word characters. A length floor (``_MIN_REDACT_LEN``) keeps short, 

314 common values from corrupting unrelated text. 

315 

316 Values are redacted LONGEST-FIRST so a shorter credential that is a 

317 prefix/substring of a longer one cannot fragment the longer value and 

318 leak its tail (e.g. token ``abcd1234`` and ``abcd1234efgh5678`` -- if 

319 the short one ran first it would split the long one, leaving 

320 ``efgh5678`` in cleartext). 

321 """ 

322 if not text: 

323 return text 

324 secrets: list[str] = [] 

325 for name, value in os.environ.items(): 

326 if not value or len(value) < _MIN_REDACT_LEN or not _matches_credential(name): 

327 continue 

328 secrets.append(value) 

329 # subprocess text=True runs in universal-newline mode, which rewrites 

330 # CRLF and lone CR to LF in captured stdout/stderr. A credential whose 

331 # value carries a carriage return (a CRLF-sourced .env var, a Windows 

332 # PEM/base64 blob) therefore diverges from this raw os.environ needle, 

333 # so the exact str.replace below would miss and leak the cleartext to 

334 # scripts.log. Mask the newline-normalized form too (same transform the 

335 # subprocess applies); keep the raw form so the command/target string, 

336 # which is never newline-translated, still matches. 

337 normalized = value.replace("\r\n", "\n").replace("\r", "\n") 

338 if normalized != value and len(normalized) >= _MIN_REDACT_LEN: 

339 secrets.append(normalized) 

340 redacted = text 

341 for value in sorted(set(secrets), key=len, reverse=True): 

342 redacted = redacted.replace(value, "[REDACTED]") 

343 return redacted

346def _redact_url_credentials(url: str) -> str: 

347 """Strip ``user:password@`` userinfo from a URL before it is logged. 

348 

349 Deterministic log hygiene for a field APM itself writes to 

350 ``scripts.log`` (the http event target): the URL is parsed with 

351 ``urllib`` and the authority rebuilt from host (+ port) only. This is 

352 NOT a shape-scan of script output -- it sanitizes one APM-owned value. 

353 The credential-bearing URL is still used for the actual dispatch; only 

354 the LOGGED form is sanitized. 

355 """ 

356 try: 

357 parts = urlsplit(url) 

358 except ValueError: 

359 return url 

360 if "@" not in (parts.netloc or ""): 

361 return url 

362 try: 

363 host = parts.hostname or "" 

364 port = parts.port 

365 except ValueError: 

366 return url 

367 if host and ":" in host and not host.startswith("["): 367 ↛ 368line 367 didn't jump to line 368 because the condition on line 367 was never true

368 host = f"[{host}]" 

369 netloc = f"{host}:{port}" if port is not None else host 

370 return urlunsplit(parts._replace(netloc=netloc))

385# ``proc.communicate()`` would otherwise buffer the ENTIRE child output (a

426def _neutralize_newlines(text: str) -> str: 

427 """Escape every line-boundary code point so a log field cannot forge a line. 

428 

429 The scripts.log audit trail exists to catch malicious scripts, so a field 

430 derived from attacker-controlled stdout/stderr (or a multi-line command)

482def _rotate_log_if_large(log_path: Path) -> None:

519 # re-stat below makes skipping safe: a later appender re-stats and
