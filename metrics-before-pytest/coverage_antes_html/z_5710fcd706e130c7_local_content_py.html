1"""Local-content integration: deploy primitives the user authored locally. 

2 

3This module handles two related scenarios: 

4 

51. **Root project as implicit local package (#714)** -- when the project's own 

6 ``.apm/`` directory contains skills, instructions, agents, prompts, hooks, 

7 or commands, ``apm install`` deploys them to target directories exactly like 

8 dependency primitives. ``_project_has_root_primitives`` and 

9 ``_has_local_apm_content`` detect this case. 

10 

112. **Local-path dependencies from apm.yml** -- ``_copy_local_package`` copies 

12 a locally-referenced package into ``apm_modules/`` so the downstream 

13 integration pipeline can treat it uniformly. 

14 

15The orchestrator ``_integrate_local_content`` lives in 

16``apm_cli.install.services`` (the DI seam) and is re-exported from 

17``apm_cli.commands.install`` for backward-compatible patching. Tests should 

18patch the symbol at the import path used by the code under test rather than 

19assuming the implementation lives in the commands module. 

20 

21Functions 

22--------- 

23_project_has_root_primitives 

24 Return True when the project root contains a ``.apm/`` directory. 

25_has_local_apm_content 

26 Return True when ``.apm/`` contains at least one primitive file. 

27_copy_local_package 

28 Copy a local-path dependency into ``apm_modules/``. 

29"""

45def _project_has_root_primitives(project_root) -> bool: 

46 """Return True when *project_root* has a .apm/ directory of its own. 

47 

48 Used to decide whether ``apm install`` should enter the integration 

49 pipeline even when no external APM dependencies are declared (#714). 

50 The integrators themselves determine whether the directory contains 

51 anything actionable, so we only check for the directory's existence. 

52 """ 

53 from pathlib import Path as _Path 

54 

55 root = _Path(project_root) 

56 return (root / ".apm").is_dir()

59def _has_local_apm_content(project_root): 

60 """Check if the project has local .apm/ content worth integrating. 

61 

62 Returns True if .apm/ exists and contains at least one primitive file 

63 in a recognized subdirectory (skills, instructions, agents/chatmodes, 

64 prompts, hooks, commands). 

65 """ 

66 apm_dir = project_root / ".apm" 

67 if not apm_dir.is_dir(): 

68 return False 

69 _PRIMITIVE_DIRS = ( 

70 "skills", 

71 "instructions", 

72 "chatmodes", 

73 "agents", 

74 "prompts", 

75 "hooks", 

76 "commands", 

77 "extensions", 

78 ) 

79 for subdir_name in _PRIMITIVE_DIRS: 

80 subdir = apm_dir / subdir_name 

81 if subdir.is_dir() and any(p.is_file() for p in subdir.rglob("*")): 

82 return True 

83 return False

91def _copy_tree_dereferencing_validated( 

92 src: Path, dst: Path, pkg_root: Path, *, _visited: set[Path] | None = None 

93) -> None: 

94 """Recursively copy *src* into *dst*, dereferencing in-package symlinks. 

95 

96 For every entry in *src*: 

97 

98 * Regular file -> ``shutil.copy2`` to *dst*. 

99 * Symlink -> resolve per-file, validate that the resolved target 

100 stays within *pkg_root* via ``ensure_path_within``, 

101 then ``shutil.copy2`` the resolved content (or recurse 

102 for symlinked directories). A symlink whose resolved 

103 target escapes *pkg_root* raises 

104 :class:`~apm_cli.utils.path_security.PathTraversalError` 

105 immediately, aborting the copy. 

106 * Directory -> recurse. 

107 

108 Each symlink undergoes a per-file resolve -> validate -> copy2 sequence 

109 rather than a bulk copytree that resolves and copies in separate passes, 

110 so the containment check fires before any filesystem write for that entry. 

111 The containment check is performed against ``pkg_root`` (the top-level 

112 package directory passed to ``_copy_local_package``) so that symlinks 

113 that cross sub-directory boundaries but remain inside the package are 

114 accepted, while any symlink that escapes the package boundary is rejected 

115 with a hard failure. 

116 

117 The ``_visited`` set tracks resolved directory paths already entered. 

118 Circular directory-symlink chains (A -> B -> A) are detected deterministically 

119 without relying on the OS ELOOP limit (which is platform-dependent). 

120 

121 A directory that cannot be listed (e.g. ``PermissionError`` from 

122 ``iterdir``) hard-fails the install with a :class:`PathTraversalError` 

123 rather than leaking a bare ``OSError`` up the install stack. 

124 """ 

125 if _visited is None: 

126 _visited = set() 

127 

128 dst.mkdir(parents=True, exist_ok=True) 

129 

130 try: 

131 entries = sorted(src.iterdir()) 

132 except OSError as exc: 

133 raise PathTraversalError( 

134 f"Cannot read package directory '{src}': {exc}. " 

135 f"Check that the package directory has read permissions. " 

136 f"Local install aborted." 

137 ) from exc 

138 

139 for entry in entries: 

140 dst_entry = dst / entry.name 

141 

142 if entry.is_symlink(): 

143 try: 

144 resolved = entry.resolve(strict=True) 

145 except OSError as exc: 

146 raise PathTraversalError( 

147 f"Symlink '{entry}' has a broken or unresolvable target: {exc}. " 

148 f"Remove the dangling symlink or restore its target. " 

149 f"Local install aborted." 

150 ) from exc 

151 

152 try: 

153 ensure_path_within(resolved, pkg_root) 

154 except PathTraversalError as _exc: 

155 raise PathTraversalError( 

156 f"Symlink '{entry}' resolves to '{resolved}', which escapes " 

157 f"the package root '{pkg_root}'. " 

158 f"Only in-package symlinks are dereferenced. " 

159 f"Remove the symlink or replace it with a real file copy. " 

160 f"Local install aborted." 

161 ) from _exc 

162 

163 if resolved.is_dir(): 

164 if resolved in _visited: 

165 raise PathTraversalError( 

166 f"Circular symlink detected: '{entry}' resolves to '{resolved}', " 

167 f"which was already visited during this install. " 

168 f"Break the cycle by replacing one symlink with a real directory copy. " 

169 f"Local install aborted." 

170 ) 

171 _copy_tree_dereferencing_validated( 

172 resolved, dst_entry, pkg_root, _visited=_visited | {resolved} 

173 ) 

174 shutil.copystat(resolved, dst_entry) 

175 else: 

176 shutil.copy2(resolved, dst_entry) 

177 elif entry.is_dir(): 

178 _copy_tree_dereferencing_validated(entry, dst_entry, pkg_root, _visited=_visited) 

179 shutil.copystat(entry, dst_entry) 

180 else: 

181 shutil.copy2(entry, dst_entry)

184def _copy_local_package(dep_ref, install_path, base_dir, *, project_root, logger): 

185 """Copy a local package to apm_modules/. 

186 

187 Args: 

188 dep_ref: DependencyReference with is_local=True. 

189 install_path: Target path under apm_modules/. 

190 base_dir: Directory used to resolve a relative ``dep_ref.local_path``. 

191 For direct deps from the root project this is the project root; 

192 for transitive deps it is the source directory of the package 

193 whose apm.yml declared *dep_ref* (#857). Must NOT be confused 

194 with ``project_root`` -- the anchoring base and the security 

195 containment boundary are deliberately distinct concerns. 

196 project_root: Project root, threaded through for symmetry with the 

197 anchoring story but NOT used as a hard containment boundary 

198 here. The actual untrusted-source boundary lives upstream in 

199 :mod:`apm_cli.deps.apm_resolver` (remote-parent local paths are 

200 expanded to same-repo virtual deps or rejected before this 

201 function ever runs). Enforcing project-root containment 

202 *here* would also block legitimate sibling layouts -- e.g. 

203 ``apm install ../pkg-a`` from a monorepo workspace -- which 

204 users explicitly opt into. Kept on the signature so callers 

205 keep the security model in mind and so a future tightening 

206 (e.g. opt-in strict mode) has a hook. 

207 logger: Required CommandLogger for structured output. Callers must 

208 thread one in; we no longer fall back to a bare console helper 

209 (#940 SR4) because doing so masked logger threading bugs in 

210 transitive call stacks. 

211 

212 Returns: 

213 install_path on success, None on failure. 

214 

215 Notes: 

216 We deliberately do NOT call ``validate_path_segments`` on 

217 ``dep_ref.local_path``: that helper rejects ``..`` segments, which 

218 would break the legitimate ``../sibling`` pattern this PR enables. 

219 The untrusted-source boundary is the resolver-level expansion or 

220 rejection of remote-parent local_paths; everything reaching this 

221 function comes from a parent the user already trusts (their own 

222 manifest, a CLI arg, or another local package they explicitly added). 

223 """ 

224 # project_root retained on signature for future strict-mode hook (see 

225 # docstring); not consumed in the current copy path. 

226 _ = project_root 

227 

228 # PR #1111 review C1: ``ctx.logger`` is allowed to be None 

229 # (``run_install_pipeline(logger=None)`` is a public, documented entry 

230 # point). Without this guard the unconditional ``logger.error(...)`` 

231 # calls below would AttributeError for any local dep when a caller 

232 # does not thread an InstallLogger through. Defaulting to the rich- 

233 # console-backed ``NullCommandLogger`` keeps the error visible to the 

234 # user while preserving the documented "logger is required" contract 

235 # for callers that DO thread one in (their logger wins). 

236 if logger is None: 

237 from apm_cli.core.null_logger import NullCommandLogger 

238 

239 logger = NullCommandLogger() 

240 

241 local = Path(dep_ref.local_path).expanduser() 

242 # Anchor on the *declaring* package's directory (#857). For direct deps 

243 # from the root, ``base_dir`` IS ``project_root`` so behavior is 

244 # unchanged. For transitive deps, ``base_dir`` is the parent package's 

245 # source dir. Absolute paths bypass anchoring. 

246 if not local.is_absolute(): # noqa: SIM108 

247 local = (base_dir / local).resolve() 

248 else: 

249 local = local.resolve() 

250 

251 if not local.is_dir(): 

252 logger.error(f"Local package path does not exist: {dep_ref.local_path}") 

253 return None 

254 from apm_cli.deps._shared import has_marketplace_deployable_manifest 

255 from apm_cli.utils.helpers import find_plugin_json 

256 

257 if ( 

258 not (local / "apm.yml").exists() 

259 and not (local / "SKILL.md").exists() 

260 and find_plugin_json(local) is None 

261 and not has_marketplace_deployable_manifest(dep_ref) 

262 ): 

263 logger.error( 

264 "Local package is not a valid APM package " 

265 f"(no apm.yml, SKILL.md, or plugin.json): {dep_ref.local_path}" 

266 ) 

267 return None 

268 

269 # Ensure parent exists and clean target (always re-copy for local deps) 

270 install_path.parent.mkdir(parents=True, exist_ok=True) 

271 if install_path.exists(): 

272 # install_path is already validated by get_install_path() (Layer 2), 

273 # but use safe_rmtree for defense-in-depth. 

274 apm_modules_dir = install_path.parent.parent # _local/<name> -> apm_modules 

275 safe_rmtree(install_path, apm_modules_dir) 

276 

277 # Dereference in-package symlinks per-file (fix for #1668). 

278 # Each symlink is resolved -> validated within `local` -> copied as a real 

279 # file, matching the behavior of the remote install path which uses 

280 # robust_copytree with symlinks=False. Symlinks that escape the package 

281 # root raise PathTraversalError immediately (hard-fail, not warn-and-skip). 

282 # On any PathTraversalError, clean up install_path to prevent dangling 

283 # partial-copy content from a malicious package. 

284 try: 

285 _copy_tree_dereferencing_validated(local, install_path, local) 

286 except PathTraversalError: 

287 if install_path.exists(): 287 ↛ 289line 287 didn't jump to line 289 because the condition on line 287 was always true

288 safe_rmtree(install_path, install_path.parent.parent) 

289 raise 

290 return install_path
