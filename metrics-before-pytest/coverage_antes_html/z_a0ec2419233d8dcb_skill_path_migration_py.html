1"""Lockfile-aware auto-migration of legacy per-client skill paths. 

2 

3When a skill's deploy target changes from a per-client directory (e.g. 

4``.github/skills/``, ``.cursor/skills/``) to the converged ``.agents/skills/`` 

5directory, the lockfile still records the old paths. This module detects those 

6stale legacy entries, checks for collisions with the new converged location, 

7and deletes the old files after the integrate phase has written the new ones. 

8 

9**Write-order safety** (crash-recovery friendly): 

10 

111. Integrate phase writes new files to ``.agents/skills/…``. 

122. This module deletes old per-client files. 

133. Lockfile phase persists the updated ``deployed_files``. 

14 

15If a crash occurs between steps 1 and 2, the next install re-detects the 

16legacy paths (still in the lockfile) and retries deletion. If a crash occurs 

17between steps 2 and 3, the lockfile still has the old paths but the files are 

18gone -- the next install sees them missing and the cleanup is a no-op. 

19 

20**Security model**: 

21 

22- Plan creation (:func:`detect_legacy_skill_deployments`) rejects lockfile 

23 entries containing path-traversal segments (``..``) via 

24 :func:`~apm_cli.utils.path_security.validate_path_segments`. 

25- File deletion (:func:`execute_migration`) guards ``unlink()`` with 

26 :func:`~apm_cli.utils.path_security.ensure_path_within` so that even a 

27 poisoned plan entry cannot delete outside the project root. 

28- Directory deletion is routed through :func:`~apm_cli.utils.path_security.safe_rmtree` 

29 which also calls ``ensure_path_within``. 

30- Parent-directory cleanup (:func:`_cleanup_empty_parents`) is 

31 containment-checked at every iteration step. 

32"""

112def detect_legacy_skill_deployments( 

113 lockfile: LockFile, 

114 project_root: Path, 

115) -> list[MigrationPlan]: 

116 """Scan the lockfile for deployed_files that match legacy per-client skill paths. 

117 

118 Returns a list of :class:`MigrationPlan` entries, one per file that should 

119 be migrated. The caller decides whether to act on the plan. 

120 

121 The *project_root* parameter is accepted for forward-compatibility (e.g. 

122 on-disk collision detection) but is unused in the initial implementation. 

123 """ 

124 plans: list[MigrationPlan] = [] 

125 

126 for dep_key, dep in lockfile.dependencies.items(): 

127 for rel_path in dep.deployed_files: 

128 m = _LEGACY_SKILL_PATTERN.match(rel_path) 

129 if not m: 

130 continue 

131 

132 # B1: reject path-traversal in lockfile-recorded paths. 

133 try: 

134 validate_path_segments(rel_path, context="legacy skill path") 

135 except PathTraversalError: 

136 _log.warning("Skipping legacy path with traversal segments: %s", rel_path) 

137 continue 

138 

139 # Build the converged destination: .agents/skills/<name>/<rest> 

140 # e.g. ".github/skills/my-skill/SKILL.md" -> ".agents/skills/my-skill/SKILL.md" 

141 client_prefix = f".{m.group(1)}/skills/" 

142 suffix = rel_path[len(client_prefix) :] 

143 

144 # B1: also validate the computed destination suffix. 

145 try: 

146 validate_path_segments(suffix, context="migrated skill suffix") 

147 except PathTraversalError: 

148 _log.warning("Skipping legacy path whose suffix has traversal: %s", rel_path) 

149 continue 

150 

151 dst = f".agents/skills/{suffix}" 

152 plans.append(MigrationPlan(src_path=rel_path, dst_path=dst, dep_name=dep_key)) 

153 

154 return plans

157def check_collisions( 

158 plans: list[MigrationPlan], 

159 project_root: Path, 

160) -> list[str]: 

161 """Check whether any migration destination already exists AND differs. 

162 

163 Returns a list of human-readable collision descriptions. An empty list 

164 means no collisions -- safe to proceed. 

165 

166 If the destination file already exists with identical content, that is NOT 

167 a collision (the integrate phase already wrote it). 

168 """ 

169 collisions: list[str] = [] 

170 for plan in plans: 

171 src_abs = project_root / plan.src_path 

172 dst_abs = project_root / plan.dst_path 

173 if not dst_abs.exists(): 

174 # Destination doesn't exist yet -- the integrate phase will create 

175 # it, or the file has already been cleaned up. Not a collision. 

176 continue 

177 if not src_abs.exists(): 

178 # Source is already gone -- nothing to collide with. 

179 continue 

180 # Both exist -- compare content. 

181 try: 

182 if src_abs.read_bytes() == dst_abs.read_bytes(): 

183 continue # Identical content: not a real collision. 

184 except OSError: 

185 pass # Fall through to report collision on I/O error. 

186 collisions.append(f"{plan.src_path} -> {plan.dst_path} (dep: {plan.dep_name})") 

187 return collisions

190def execute_migration( 

191 plans: list[MigrationPlan], 

192 lockfile: LockFile, 

193 project_root: Path, 

194) -> MigrationResult: 

195 """Delete legacy files and update lockfile ``deployed_files`` in place. 

196 

197 For each plan entry: 

198 

199 1. Delete the legacy file from disk via :func:`safe_rmtree`. 

200 2. Remove the old path from ``deployed_files`` on the owning dependency. 

201 3. Add the new ``.agents/skills/`` path if not already present. 

202 4. Mirror changes in ``local_deployed_files`` / ``local_deployed_file_hashes`` 

203 when the path originates from the synthesized self-entry (local bundles). 

204 

205 The lockfile object is mutated in place; the caller is responsible for 

206 persisting it to disk afterwards. 

207 

208 Returns a :class:`MigrationResult` summarizing what happened. 

209 """ 

210 result = MigrationResult() 

211 

212 for plan in plans: 

213 abs_path = project_root / plan.src_path 

214 

215 if not abs_path.exists(): 

216 result.skipped_no_file.append(plan.src_path) 

217 # Still update lockfile even if file is gone -- the path entry 

218 # is stale and should point to the new location. 

219 _update_lockfile_entry(lockfile, plan, result) 

220 continue 

221 

222 try: 

223 # B1: containment guard before any disk mutation. 

224 ensure_path_within(abs_path, project_root) 

225 if abs_path.is_file(): 225 ↛ 228line 225 didn't jump to line 228 because the condition on line 225 was always true

226 abs_path.unlink() 

227 else: 

228 safe_rmtree(abs_path, project_root) 

229 result.deleted.append(plan.src_path) 

230 _cleanup_empty_parents(abs_path, project_root) 

231 except (OSError, PathTraversalError): 

232 result.failed.append(plan.src_path) 

233 continue # Don't update lockfile for failed deletions. 

234 

235 _update_lockfile_entry(lockfile, plan, result) 

236 

237 if result.updated_deps: 

238 from apm_cli.core.deployment_ledger import DeploymentLedgerCodec 

239 

240 DeploymentLedgerCodec.refresh_from_legacy(lockfile) 

241 return result

252 ``local_deployed_file_hashes`` when the path appears there (local 

253 bundles use the flat fields as source of truth for serialization). 

254 """ 

255 dep = lockfile.dependencies.get(plan.dep_name) 

256 if dep is None: 256 ↛ 257line 256 didn't jump to line 257 because the condition on line 256 was never true

257 return 

258

261 files.remove(plan.src_path) 

262 if plan.dst_path not in files:

266 hashes = dep.deployed_file_hashes

273 lockfile.rename_local_deployed_path(plan.src_path, plan.dst_path)
