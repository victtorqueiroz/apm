16# Hard byte ceiling for an MCP registry HTTP body. The registry URL can come 

17# from an untrusted cloned apm.yml (``MCPDependency.registry``), so the 

18# discovery client must bound the body BEFORE buffering it -- a multi-GB body 

19# would otherwise exhaust memory on ``apm install``. Mirrors the round-26 cap 

20# on the sibling ``deps/registry/client.py``. 

21_MAX_REGISTRY_JSON_BYTES = 10 * 1024 * 1024 # 10 MiB 

22_REGISTRY_CHUNK_BYTES = 64 * 1024 

23# Decode failures that must fail CLOSED (a deep-nested body raises 

24# ``RecursionError``, an oversized int ``ValueError``, non-UTF-8 

25# ``UnicodeDecodeError``) rather than escape the client raw. 

26_REGISTRY_JSON_ERRORS = (ValueError, json.JSONDecodeError, RecursionError, UnicodeDecodeError)

772 def _is_server_match(self, reference: str, server_name: str) -> bool: 

773 """Check if a reference matches a server name using common patterns. 

774 

775 Matching rules: 

776 1. Exact string match always wins. 

777 2. Qualified references (contain '/') match if the server name ends 

778 with the reference (e.g. 'github/github-mcp-server' matches 

779 'io.github.github/github-mcp-server'). The match must happen at 

780 a namespace boundary (preceded by '.' or start-of-string) to 

781 prevent slug collisions like 'microsoftdocs/mcp' matching 

782 'com.supabase/mcp'. 

783 3. Unqualified references fall back to slug (last segment) comparison. 

784 

785 Args: 

786 reference (str): Original reference from user. 

787 server_name (str): Server name from registry. 

788 

789 Returns: 

790 bool: True if they represent the same server. 

791 """ 

792 # Direct match 

793 if reference == server_name: 

794 return True 

795 

796 if "/" in reference: 

797 # Qualified reference: allow suffix match at a namespace boundary. 

798 # e.g. "github/github-mcp-server" matches "io.github.github/github-mcp-server" 

799 # but "microsoftdocs/mcp" must NOT match "com.supabase/mcp". 

800 if server_name.endswith(reference): 

801 prefix = server_name[: -len(reference)] 

802 # Valid boundary: empty (exact), or ends with '.' (namespace separator) 

803 if prefix == "" or prefix.endswith("."): 

804 return True 

805 return False 

806 

807 # Unqualified reference: fall back to slug comparison 

808 ref_repo = self._extract_repository_name(reference) 

809 server_repo = self._extract_repository_name(server_name) 

810 

811 return ref_repo == server_repo
