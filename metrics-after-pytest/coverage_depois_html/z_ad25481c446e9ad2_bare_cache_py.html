60def _scrub_bare_remote_url(bare_path: Path, git_exe: str, env: dict[str, str]) -> None: 

61 """Redact ``remote.origin.url`` in a bare repo's ``.git/config``. 

62 

63 After a successful bare clone, ``remote.origin.url`` retains the 

64 tokenized URL (e.g. ``https://oauth2:<TOKEN>@github.com/...``). The 

65 bare is read-only after this point in the WS2 dedup pipeline (no 

66 further fetches), so the URL is dead weight that just persists the 

67 token on disk. Replace with ``redacted://`` to eliminate the 

68 on-disk token footprint. 

69 

70 Defense-in-depth: tier-1 (init + remote add + fetch) leaves 

71 ``FETCH_HEAD`` containing the tokenized URL on disk even after the 

72 config scrub. Truncate it to empty so the token does not survive 

73 in any on-disk artifact. Best-effort (non-fatal on OSError). 

74 

75 Best-effort: ``check=False`` so a config-set failure does not abort 

76 the clone (the bare is still functionally correct without the 

77 redaction). Convergent panel finding (auth + supply-chain MAJOR). 

78 On exception, log at WARNING so token-leak-aware operators have an 

79 audit trail (supply-chain reviewer follow-up: security mechanisms 

80 must not fail silently). 

81 """ 

82 try: 

83 result = subprocess.run( 

84 [git_exe, "--git-dir", str(bare_path), "remote", "set-url", "origin", "redacted://"], 

85 env=env, 

86 check=False, 

87 capture_output=True, 

88 timeout=10, 

89 ) 

90 if result.returncode != 0: 

91 logging.getLogger(__name__).warning( 

92 "Failed to redact remote URL from bare repo config at %s " 

93 "(git exit=%d). Tokenized URL may persist on disk until " 

94 "shared cache cleanup.", 

95 bare_path, 

96 result.returncode, 

97 ) 

98 except Exception as exc: 

99 logging.getLogger(__name__).warning( 

100 "Exception while redacting remote URL from bare repo config " 

101 "at %s: %s. Tokenized URL may persist on disk until shared " 

102 "cache cleanup.", 

103 bare_path, 

104 exc, 

105 ) 

106 

107 # Defense-in-depth: truncate FETCH_HEAD which retains the tokenized 

108 # URL after tier-1 init+fetch (supply-chain panel follow-up). 

109 fetch_head = bare_path / "FETCH_HEAD" 

110 try: 

111 if fetch_head.exists(): 

112 fetch_head.write_text("") 

113 except OSError as exc: 

114 logging.getLogger(__name__).warning(

132 

133 Sibling helper to :meth:`GitHubPackageDownloader._clone_with_fallback`. 

134 Composes via the caller-supplied ``execute_transport_plan`` callable 

135 (typically ``self._execute_transport_plan``) so it inherits ADO 

136 bearer retry and protocol fallback automatically. The bare clone is 

137 subdir-agnostic (no sparse cone), so a single bare can serve N 

138 consumers each materializing a different subdir from the same 

139 repo+ref - this is the core fix for #1126. 

140 

141 3-tier strategy (per design.md sec 5.2): 

142 

143 - Tier 1 (SHA refs): ``git init --bare && git remote add 

144 origin <url> && git fetch --depth=1 origin <sha>``. Requires 

145 server-side ``uploadpack.allowReachableSHA1InWant=true`` 

146 (default on github.com / GHES; some older GHE / ADO Server / 

147 Bitbucket Server reject this). 

148 - Tier 1 (symbolic / default branch): ``git clone --bare 

149 --depth=1 [--branch <ref>] <url>``. 

150 - Tier 2 (both): full ``git clone --bare <url>``, validate via 

151 ``git rev-parse --verify <sha>^{commit}`` for SHA refs. 

152 - Tier 3: re-raise. 

153 

154 After every successful tier, ``git update-ref HEAD <ref>`` is 

155 invoked for SHA refs so consumer ``git rev-parse HEAD`` resolves 

156 unambiguously (the v3 BLOCKER fix). Then 

157 :func:`_scrub_bare_remote_url` redacts the tokenized 

158 ``remote.origin.url`` from ``.git/config`` to eliminate 

159 on-disk token persistence (panel convergent finding). 

160

162 == known_sha``) is intentionally deferred. The bare is 

163 ephemeral, mode-0700, and produced by a trusted (in-tree) 

164 callable. If ``SharedCloneCache`` is ever opened to 

165 plugin/user-supplied callables, restore the integrity check at 

166 the cache boundary. See design.md sec 12 (Bare integrity 

167 verification).

316def fetch_sha_into_bare( 

317 execute_transport_plan: Callable[..., None], 

318 repo_url_base: str, 

319 bare_path: Path, 

320 sha: str, 

321 *, 

322 dep_ref: DependencyReference, 

323) -> bool: 

324 """Attempt to fetch a specific SHA into an existing bare repo. 

325 

326 Used to hydrate shallow bare clones that are missing a transitive 

327 SHA-pinned commit. Three-step strategy: 

328 

329 1. **Check first** -- ``git rev-parse --verify <sha>^{commit}`` against 

330 the bare. If the SHA is already present, returns ``True`` immediately 

331 without any network I/O. 

332 2. **Shallow fetch by SHA** (full 40-char SHAs only) -- invokes 

333 ``execute_transport_plan`` with a fetch action that runs 

334 ``git fetch <url> <sha>``. Uses the authenticated URL supplied by 

335 the transport plan, NOT ``git fetch origin <sha>``, because 

336 ``remote.origin.url`` has been redacted to ``redacted://`` by 

337 :func:`_scrub_bare_remote_url`. After the fetch, verifies with 

338 ``rev-parse --verify``. Returns ``True`` on success. 

339 3. **Broaden shallow** -- invokes ``execute_transport_plan`` with a 

340 fetch action that runs ``git fetch <url>`` (no ref argument), 

341 broadening the shallow boundary to include all remote refs. After 

342 the fetch, verifies with ``rev-parse --verify``. Returns ``True`` 

343 on success. 

344 

345 On any failure in steps 2 or 3, returns ``False`` so the caller can 

346 fall back to a fresh bare clone. 

347 

348 Note: this function deliberately does NOT call ``git update-ref HEAD`` 

349 after a successful fetch. The consumer's :func:`materialize_from_bare` 

350 handles SHA resolution independently via the ``known_sha`` parameter. 

351 

352 Args: 

353 execute_transport_plan: Callable that orchestrates auth and protocol 

354 fallback (typically ``self._execute_transport_plan``). 

355 repo_url_base: Base repo URL (unauthenticated) passed to the 

356 transport plan so it can inject credentials. 

357 bare_path: Path to the existing bare repo on disk. 

358 sha: The Git commit SHA to fetch. 

359 dep_ref: Dependency reference used by the transport plan for 

360 auth context. 

361 

362 Returns: 

363 ``True`` if the SHA is now present in the bare, ``False`` otherwise. 

364 """

391 def _pin_sha_as_head_ref() -> None: 

392 """Add refs/heads/apm-pin-<sha-prefix> so the SHA is reachable via git-clone. 

393 

394 ``git clone --local --shared`` from a *shallow* bare ignores 

395 ``--shared`` and falls back to the upload-pack protocol, which 

396 only transfers objects reachable from advertised refs. A SHA 

397 fetched by :func:`fetch_sha_into_bare` is inserted into the 

398 object store but is *not* referenced by any ref, so the clone 

399 silently omits it and subsequent ``git checkout <sha>`` fails. 

400 

401 Creating a synthetic ``refs/heads/apm-pin-*`` ref makes the 

402 commit reachable via the default ``refs/heads/*`` refspec, so 

403 upload-pack includes it. Best-effort: a failure here is logged 

404 at DEBUG level and does not abort the install (the fallback is 

405 a fresh bare clone for the pinned package). 

406 """

561def materialize_from_bare( 

562 bare_path: Path, 

563 consumer_dir: Path, 

564 *, 

565 ref: str | None, 

566 env: dict[str, str], 

567 known_sha: str | None = None, 

568 sparse_paths: list[str] | None = None, 

569) -> str: 

570 """Create a working-tree checkout from a bare repo via local-shared clone. 

571 

572 Mirrors :class:`GitCache`'s ``_create_checkout`` pattern: each 

573 consumer gets its own working tree backed by the shared bare's 

574 object database (via ``objects/info/alternates``). Hardlink-cheap 

575 and concurrent-safe (consumer dirs are unique per call). 

576 

577 SHA resolution policy (lifetime invariant 5.2.1): 

578 - If ``known_sha`` is provided (caller passed a 40-char SHA 

579 ref), use it directly. Avoids ``rev-parse HEAD`` which is 

580 ambiguous on init+fetch bares before update-ref runs. 

581 - Otherwise, resolve from the BARE via ``git --git-dir 

582 <bare> rev-parse HEAD``. NOT from the consumer - opening 

583 ``Repo(consumer_dir)`` leaks a Windows file handle that 

584 blocks downstream rmtree. 

585 

586 CRLF + LFS pinning before checkout: 

587 - ``core.autocrlf=false`` guarantees byte-identical content 

588 across consumers regardless of the user's global git config. 

589 - ``filter.lfs.smudge=""`` + ``filter.lfs.required=false`` 

590 disables LFS smudge cross-platform (the empty string trick 

591 works everywhere; ``cat`` is not on Windows PATH). 

592 

593 Sparse-cone (perf #1433): 

594 - When ``sparse_paths`` is set, ``git sparse-checkout init --cone`` 

595 followed by ``git sparse-checkout set <paths...>`` runs BEFORE 

596 ``git checkout``, so only the requested top-level directories 

597 are materialized in the working tree. The bare's object DB is 

598 unchanged; only the consumer working tree shrinks. 

599 - Sparse-checkout failures are RAISED (not silently fallen back) 

600 because a silent fallback would re-introduce the 78 MB bloat 

601 this parameter exists to avoid. 

602 - After checkout, if the cone left a dangling symlink (target 

603 outside the requested paths -- #2707), falls back to 

604 ``git sparse-checkout disable`` so the target resolves. This is 

605 a correctness repair, not a failure fallback: it only fires 

606 when the narrow cone would otherwise ship a broken checkout. 

607 

608 Returns: 

609 The resolved commit SHA. Caller threads this into 

610 ``resolved_commit`` for the lockfile. 

611 """ 

612 from ..utils.git_env import ( 

613 get_git_executable, 

614 git_no_hooks_args, 

615 git_no_templates_args, 

616 git_subprocess_env, 

617 ) 

618 

619 git_exe = get_git_executable() 

620 env = git_subprocess_env(env) 

621 

622 if known_sha: 

623 resolved_sha = known_sha 

624 else: 

625 sha_result = subprocess.run( 

626 [git_exe, "--git-dir", str(bare_path), "rev-parse", "HEAD"], 

627 capture_output=True, 

628 text=True, 

629 timeout=10, 

630 env=env, 

631 check=True, 

632 ) 

633 resolved_sha = sha_result.stdout.strip() 

634 

635 consumer_dir.parent.mkdir(parents=True, exist_ok=True) 

636 # --no-checkout because we want to set core.autocrlf and disable 

637 # LFS smudge BEFORE checkout writes any file content. 

638 subprocess.run( 

639 [ 

640 git_exe, 

641 *git_no_hooks_args(), 

642 "clone", 

643 *git_no_templates_args(), 

644 "--local", 

645 "--shared", 

646 "--no-checkout", 

647 str(bare_path), 

648 str(consumer_dir), 

649 ], 

650 capture_output=True, 

651 text=True, 

652 timeout=60, 

653 env=env, 

654 check=True, 

655 ) 

656 # CRLF determinism (panel: byte-identical across users). 

657 subprocess.run( 

658 [git_exe, "-C", str(consumer_dir), "config", "core.autocrlf", "false"], 

659 capture_output=True, 

660 text=True, 

661 timeout=10, 

662 env=env, 

663 check=True, 

664 ) 

665 # Disable LFS smudge cross-platform: empty-string smudge is the 

666 # portable equivalent of `git lfs smudge --skip`. The `cat` 

667 # alternative is not on Windows PATH. 

668 for key, val in ( 

669 ("filter.lfs.smudge", ""), 

670 ("filter.lfs.clean", ""), 

671 ("filter.lfs.process", ""), 

672 ("filter.lfs.required", "false"), 

673 ): 

674 subprocess.run( 

675 [git_exe, "-C", str(consumer_dir), "config", key, val], 

676 capture_output=True, 

677 text=True, 

678 timeout=10, 

679 env=env, 

680 check=False, 

681 ) 

682 checkout_target = known_sha or "HEAD" 

683 if sparse_paths: 

684 apply_sparse_cone(git_exe, consumer_dir, list(sparse_paths), env=env) 

685 subprocess.run( 

686 [ 

687 git_exe, 

688 *git_no_hooks_args(), 

689 "-C", 

690 str(consumer_dir), 

691 "checkout", 

692 checkout_target, 

693 ], 

694 capture_output=True, 

695 text=True, 

696 timeout=60, 

697 env=env, 

698 check=True, 

699 ) 

700 if sparse_paths: 

701 dangling = repair_dangling_cone_symlinks(git_exe, consumer_dir, list(sparse_paths), env=env) 

702 if dangling is not None: 

703 _log.info( 

704 "Sparse-cone checkout of %s left a dangling symlink at %s " 

705 "(target outside the requested cone); widened to a full " 

706 "checkout so it resolves (#2707).", 

707 consumer_dir, 

708 dangling, 

709 ) 

710 return resolved_sha

783def _git_failure_text(error: Exception) -> str: 

784 """Return captured command text for internal marker classification only. 

785 

786 The result may contain credentials, key paths, or remote-controlled text. 

787 Never surface it to users, logs, or exception messages. 

788 """ 

789 parts = [str(error)] 

790 for attr in ("stderr", "stdout"): 

791 stream = getattr(error, attr, None) 

792 if not stream: 

793 continue 

794 if isinstance(stream, bytes): 

795 parts.append(stream.decode("utf-8", errors="replace")) 

796 else: 

797 parts.append(str(stream)) 

798 return " ".join(part.strip() for part in parts if part and part.strip())
