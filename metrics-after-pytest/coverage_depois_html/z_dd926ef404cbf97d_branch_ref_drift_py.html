1"""Always-on heal: re-download when a branch ref's remote SHA has 

2advanced past the lockfile-recorded SHA. 

3 

4Without this heal, ``lockfile_match=True`` would short-circuit via the 

5content-hash fallback (typical for virtual packages where install_path 

6is not a git repo), producing a 3-way inconsistency: resolved_commit 

7advances in the lockfile, content_hash and on-disk content stay stale. 

8 

9See PR #1158 root-cause analysis. The L94 comment in 

10``_resolve_download_strategy`` ("Branches always fetch latest") 

11documented the intent; this heal is the enforcement. 

12""" 

13 

14from __future__ import annotations 

15 

16from .base import HealContext, HealMessageLevel 

17 

18 

19class BranchRefDriftHeal: 

20 name = "branch_ref_drift" 

21 order = 10 

22 exclusive_group = "branch_drift" 

23 

24 def applies(self, hctx: HealContext) -> bool: 

25 from apm_cli.models.apm_package import GitReferenceType 

26 

27 if not hctx.lockfile_match or hctx.update_refs: 

28 return False 

29 if hctx.resolved_ref is None: 

30 return False 

31 if hctx.resolved_ref.ref_type != GitReferenceType.BRANCH: 

32 return False 

33 # Guard against non-git resolution paths (e.g. Artifactory proxy 

34 # may return a ResolvedReference whose resolved_commit is None or 

35 # the "cached" sentinel). Without this guard a None comparison 

36 # below would mis-classify drift, and execute() would crash on 

37 # `resolved_commit[:8]`. 

38 remote_sha = hctx.resolved_ref.resolved_commit 

39 if remote_sha in (None, "", "cached"): 

40 return False 

41 if hctx.existing_lockfile is None: 

42 return False 

43 locked = hctx.existing_lockfile.get_dependency(hctx.package_key) 

44 if locked is None: 

45 return False 

46 if locked.resolved_commit in (None, "", "cached"): 

47 return False 

48 return remote_sha != locked.resolved_commit 

49 

50 def execute(self, hctx: HealContext) -> None: 

51 locked = hctx.existing_lockfile.get_dependency(hctx.package_key) 

52 hctx.lockfile_match = False 

53 hctx.ref_changed = True 

54 # Tell FreshDependencySource that a content_hash change for this 

55 # dep is LEGITIMATE (caused by upstream branch advancing past the 

56 # lockfile-recorded SHA), not a supply-chain attack. Without 

57 # this the supply-chain hard-block at sources.py would abort the 

58 # install before the repaired lockfile is 

59 # written. 

60 hctx.add_bypass_key(hctx.package_key) 

61 hctx.emit( 

62 HealMessageLevel.INFO, 

63 f" Branch ref drift: {hctx.package_key} remote @" 

64 f"{hctx.resolved_ref.resolved_commit[:8]} != locked @" 

65 f"{locked.resolved_commit[:8]} -- forcing re-download", 

66 )
