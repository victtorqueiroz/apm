128def _local_path_from_source(value: str) -> str: 

129 """Normalise a local-path source to an absolute path string. 

130 

131 Handles three ``file://`` shapes so Windows callers get the right answer: 

132 - POSIX: ``file:///abs/path`` -> ``/abs/path`` 

133 - Windows proper: ``file:///C:/path`` -> ``C:/path`` 

134 - Windows malformed (e.g. ``f"file://{Path}"`` where Path is ``C:\\...``): 

135 ``file://C:\\path`` -> ``C:\\path`` 

136 ``urlsplit`` mis-parses the Windows-shaped forms (treats ``C:`` as a host), 

137 so do the strip manually after detecting a drive-letter prefix. 

138 """ 

139 if value.startswith("file://"): 

140 rest = value[len("file://") :] 

141 if len(rest) >= 3 and rest[0] == "/" and rest[1].isalpha() and rest[2] == ":": 

142 return rest[1:] 

143 if len(rest) >= 2 and rest[0].isalpha() and rest[1] == ":": 

144 return rest 

145 try: 

146 parsed = urlsplit(value) 

147 except ValueError: 

148 return value 

149 return parsed.path or value 

150 # Plain path: expand ~ only. Avoid Path(...) round-trip which would 

151 # rewrite POSIX separators to backslashes on Windows. 

152 return os.path.expanduser(value)
