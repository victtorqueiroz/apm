303 Returns an empty list -- never raises -- if *apm_yml_path* is ``None``, 

304 missing, or malformed; the project's own manifest is a Step-1/3 

305 concern here, not part of this walk's fail-closed contract, which 

306 covers the manifests visited DURING the walk, not this seed step.

634 # A package arg is either: (a) a marketplace ref in 

635 # `name@marketplace[#ref]` form (no slash), (b) an `owner/repo` 

636 # slug, or (c) a local filesystem path. The legacy guard below 

637 # only handled (a) when there is no `/`, but Windows absolute 

638 # paths use backslashes (e.g. `C:\Users\...\my-pkg`) and have 

639 # no `/` either -- they were wrongly rejected as "Invalid 

640 # package format" and the DB row for any deployed copilot-app 

641 # workflow would leak. Use the canonical local-path detector 

642 # so paths fall through to DependencyReference parsing on 

643 # every platform.

648 if canonical is None: 

649 # Error already logged by _resolve_marketplace_packages; 

650 # surface in packages_not_found so caller counts are accurate. 

651 packages_not_found.append(package) 

652 continue 

653 canonical_for_match = canonical 

654 display_label = package 

655 else: 

656 logger.error( 

657 f"Invalid package format: {package}. " 

658 "Use 'owner/repo', 'plugin-name@marketplace', a local path, " 

659 "or a key copied from 'apm deps list'." 

660 ) 

661 packages_not_found.append(package) 

662 continue 

663 else: 

664 canonical_for_match = package 

665 display_label = package 

666 

667 selection = select_manifest_dependency(canonical_for_match, current_deps, lockfile) 

668 if selection.status is DependencySelectionStatus.MATCHED: 

669 matched_dep = selection.manifest_entry 

670 if matched_dep not in packages_to_remove: 

671 packages_to_remove.append(matched_dep) 

672 if not log_matches: 

673 continue 

674 if canonical_for_match != display_label: 

675 logger.progress( 

676 f"{display_label} - found in apm.yml (as {canonical_for_match})", 

677 symbol="check", 

678 ) 

679 else: 

680 logger.progress( 

681 f"{_dependency_public_label(matched_dep)} - found in apm.yml", 

682 symbol="check", 

683 ) 

684 elif selection.status is DependencySelectionStatus.AMBIGUOUS: 

685 packages_not_found.append(package) 

686 if is_local: 

687 safe_label = _dependency_public_label(package) 

688 logger.error( 

689 f"Local dependency '{safe_label}' is ambiguous: it matches " 

690 f"{selection.match_count} declarations. Use one exact path from " 

691 "apm.yml to uninstall a single dependency." 

692 ) 

693 else: 

694 logger.error( 

695 f"Ambiguous dependency '{display_label}': it matches " 

696 f"{selection.match_count} declarations. Use one exact path from apm.yml " 

697 "to uninstall a single dependency." 

698 ) 

699 else: 

700 packages_not_found.append(package) 

701 if canonical_for_match.startswith("_local/"): 

702 logger.error( 

703 f"{display_label} could not be mapped to one declared local dependency. " 

704 "Run 'apm install' to regenerate apm.lock.yaml, or use one exact " 

705 "path from apm.yml." 

706 ) 

707 elif is_local: 

708 logger.error( 

709 "The requested local path was not found in apm.yml. " 

710 "Copy a local key from 'apm deps list', or use one exact " 

711 "path already declared in apm.yml." 

712 ) 

713 elif canonical_for_match != display_label: 

714 logger.error( 

715 f"{display_label} ({canonical_for_match}) was not found in apm.yml. " 

716 "Run 'apm deps list' and retry with an installed package identifier." 

717 ) 

718 else: 

719 logger.error( 

720 f"{display_label} was not found in apm.yml. " 

721 "Run 'apm deps list' and retry with an installed package identifier." 

722 ) 

723 

724 return packages_to_remove, packages_not_found
