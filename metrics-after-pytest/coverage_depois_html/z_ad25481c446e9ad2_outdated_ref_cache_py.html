1"""Per-invocation ``git ls-remote`` dedup for ``apm outdated``. 

2 

3``apm outdated`` checks every locked dependency against its upstream by 

4calling ``downloader.list_remote_refs`` / ``list_remote_tag_refs``, each of 

5which issues a fresh ``git ls-remote`` (see 

6:class:`apm_cli.deps.git_reference_resolver.GitReferenceResolver`). When 

7several locked deps share one upstream repo -- e.g. multiple virtual 

8subdirectory packages carved out of the same monorepo -- that is one 

9redundant ``ls-remote`` per dep for identical output. 

10 

11This wrapper collapses those redundant round-trips to one per 

12(repo-identity, ref-family) within a single ``apm outdated`` invocation. It 

13is the ``outdated``-surface analogue of the install-path RefResolver reuse 

14in :mod:`apm_cli.install.helpers.ref_reuse`, extending the dedup so the 

15optimization is pervasive across install and outdated. 

16 

17Correctness: the cache is built fresh per invocation and never persisted, so 

18a newer upstream ref pushed between two ``apm outdated`` runs is still seen 

19on the next run. Within one run every same-repo dep observes one consistent 

20ls-remote snapshot, which cannot make ``outdated`` miss a newer upstream ref 

21(the snapshot is the freshest listing of this run). Failures are not cached, 

22preserving the per-dep ``unknown`` degradation of the un-wrapped path. 

23 

24Auth-lens: the cache key is the repo *identity* (host, repo_url, port, 

25insecure-transport, ref-family) -- never a token. Two deps from one repo 

26resolve the same credential via ``AuthResolver`` and the same upstream refs, 

27so identity alone is a sufficient and non-secret key. 

28""" 

29

32import threading 

33from typing import TYPE_CHECKING, Any

35if TYPE_CHECKING:

40def _identity_key(dep_ref: DependencyReference, include_heads: bool) -> tuple: 

41 """Return a non-secret cache key for *dep_ref*'s ls-remote listing. 

42 

43 ``include_heads`` distinguishes the tags+branches listing 

44 (``list_remote_refs``) from the tags-only listing 

45 (``list_remote_tag_refs``): the two return different ref sets and must 

46 not share a bucket. No token is part of the key -- see module docstring. 

47 """ 

48 return ( 

49 dep_ref.host or "", 

50 dep_ref.repo_url or "", 

51 getattr(dep_ref, "port", None), 

52 bool(getattr(dep_ref, "is_insecure", False)), 

53 include_heads, 

54 )

57class OutdatedRefCache: 

58 """Thread-safe, per-invocation ``ls-remote`` dedup around a downloader. 

59 

60 Wraps a ``GitHubPackageDownloader`` so the two ref-listing methods used 

61 by ``apm outdated`` memoize their result per repo identity while every 

62 other attribute/method delegates to the wrapped downloader unchanged. 

63 Concurrent first-touches for the same key coalesce under a per-key lock 

64 so the parallel ``-j`` path still issues exactly one ``ls-remote``. 

65 """ 

66 

67 def __init__(self, downloader: Any) -> None: 

68 self._downloader = downloader 

69 self._results: dict[tuple, list[RemoteRef]] = {} 

70 self._key_locks: dict[tuple, threading.Lock] = {} 

71 self._master_lock = threading.Lock() 

72 

73 def __getattr__(self, name: str) -> Any: 

74 # Delegate everything not overridden here (registry/marketplace paths, 

75 # tiered-resolver wiring, etc.) to the wrapped downloader untouched. 

76 return getattr(self._downloader, name) 

77 

78 def list_remote_refs(self, dep_ref: DependencyReference) -> list[RemoteRef]: 

79 return self._cached(dep_ref, include_heads=True) 

80 

81 def list_remote_tag_refs(self, dep_ref: DependencyReference) -> list[RemoteRef]: 

82 return self._cached(dep_ref, include_heads=False) 

83 

84 def _cached(self, dep_ref: DependencyReference, *, include_heads: bool) -> list[RemoteRef]: 

85 key = _identity_key(dep_ref, include_heads) 

86 with self._master_lock: 

87 cached = self._results.get(key) 

88 if cached is not None: 

89 return cached 

90 key_lock = self._key_locks.get(key) 

91 if key_lock is None: 

92 key_lock = threading.Lock() 

93 self._key_locks[key] = key_lock 

94 # Compute outside the master lock (so distinct repos fetch in 

95 # parallel) but under a per-key lock (so same-repo first-touches 

96 # coalesce to one ls-remote). Failures propagate uncached, matching 

97 # the un-wrapped per-dep 'unknown' degradation. 

98 with key_lock: 

99 cached = self._results.get(key) 

100 if cached is not None: 

101 return cached 

102 if include_heads: 

103 refs = self._downloader.list_remote_refs(dep_ref) 

104 else: 

105 refs = self._downloader.list_remote_tag_refs(dep_ref) 

106 self._results[key] = refs 

107 return refs
